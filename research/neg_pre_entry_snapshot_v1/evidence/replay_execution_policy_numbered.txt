    29	def _step(value: float, step: float, rounding: str) -> float:
    30	    value_d = Decimal(str(value))
    31	    step_d = Decimal(str(step))
    32	    units = (value_d / step_d).to_integral_value(rounding=rounding)
    33	    return float(units * step_d)
    34	
    35	
    36	def floor_to_step(value: float, step: float) -> float:
    37	    return _step(value, step, ROUND_FLOOR)
    38	
    39	
    40	def ceil_to_step(value: float, step: float) -> float:
    41	    return _step(value, step, ROUND_CEILING)
    42

    44	def build_entry_price(side: str, signal_price: float, config: ReplayConfig) -> float:
    45	    if side == "LONG":
    46	        raw = max(signal_price + config.entry_offset_usd, signal_price + config.tick_size)
    47	        return floor_to_step(raw, config.tick_size)
    48	    if side == "SHORT":
    49	        raw = min(signal_price - config.entry_offset_usd, signal_price - config.tick_size)
    50	        return ceil_to_step(raw, config.tick_size)
    51	    raise BacktestContractError(f"invalid side={side}")
    52	
    53	
    54	def notional_to_qty(entry: float, notional: float, config: ReplayConfig) -> float:
    55	    if entry <= 0:
    56	        return 0.0
    57	    return floor_to_step(notional / entry, config.qty_step)

    60	def split_quantity(qty_total: float, config: ReplayConfig) -> tuple[float, float, float]:
    61	    total_units = int((Decimal(str(qty_total)) / Decimal(str(config.qty_step))).to_integral_value(rounding=ROUND_FLOOR))
    62	    if total_units <= 0:
    63	        raise BacktestContractError("quantity rounds to zero")
    64	    u1 = total_units // 3
    65	    u2 = total_units // 3
    66	    u3 = total_units - u1 - u2
    67	    if u1 <= 0 or u2 <= 0:
    68	        u1 = total_units // 2
    69	        u2 = total_units - u1
    70	        u3 = 0
    71	    if u1 <= 0 or u2 <= 0 or u1 + u2 + u3 != total_units:
    72	        raise BacktestContractError("invalid quantity split")
    73	    step = Decimal(str(config.qty_step))
    74	    return float(Decimal(u1) * step), float(Decimal(u2) * step), float(Decimal(u3) * step)

    77	def initial_stop(
    78	    side: str,
    79	    entry: float,
    80	    pre_signal_prices: list[float],
    81	    config: ReplayConfig,
    82	) -> float:
    83	    pct_stop = entry * (1 - config.sl_pct) if side == "LONG" else entry * (1 + config.sl_pct)
    84	    if side == "LONG":
    85	        swing = (
    86	            min(pre_signal_prices) - config.initial_swing_buffer_usd
    87	            if pre_signal_prices
    88	            else pct_stop
    89	        )
    90	        stop = min(pct_stop, swing, entry - config.tick_size)
    91	        return floor_to_step(stop, config.tick_size)
    92	    if side == "SHORT":
    93	        swing = (
    94	            max(pre_signal_prices) + config.initial_swing_buffer_usd
    95	            if pre_signal_prices
    96	            else pct_stop
    97	        )
    98	        stop = max(pct_stop, swing, entry + config.tick_size)
    99	        return ceil_to_step(stop, config.tick_size)
   100	    raise BacktestContractError(f"invalid side={side}")

   103	def select_volume_confirmed_initial_stop(
   104	    side: str,
   105	    entry: float,
   106	    bars: list[FeedBar],
   107	    config: ReplayConfig,
   108	) -> InitialSwingSelection:
   109	    """Select the highest-volume confirmed swing whose buffered stop is within cap.
   110	
   111	    A swing is confirmed entirely before the signal cutoff using strict fractal
   112	    comparisons over ``initial_swing_lr`` bars on each side. Synthetic bars cannot
   113	    define or confirm a swing. Ties on volume prefer the most recent swing.
   114	    """
   115	
   116	    lr = config.initial_swing_lr
   117	    if len(bars) < 2 * lr + 1:
   118	        raise InitialStopSelectionError(
   119	            "NO_FULL_INITIAL_SWING_WINDOW",
   120	            f"bars={len(bars)} required={2 * lr + 1}",
   121	        )
   122	
   123	    confirmed_count = 0
   124	    eligible: list[tuple[float, datetime, float, float]] = []
   125	    for index in range(lr, len(bars) - lr):
   126	        neighborhood = bars[index - lr:index + lr + 1]
   127	        if any(bar.is_synthetic for bar in neighborhood):
   128	            continue
   129	        bar = bars[index]
   130	        if side == "LONG":
   131	            swing_price = bar.low
   132	            if swing_price >= entry:
   133	                continue
   134	            left = [item.low for item in bars[index - lr:index]]
   135	            right = [item.low for item in bars[index + 1:index + lr + 1]]
   136	            is_swing = all(swing_price < value for value in left + right)
   137	        elif side == "SHORT":
   138	            swing_price = bar.high
   139	            if swing_price <= entry:
   140	                continue
   141	            left = [item.high for item in bars[index - lr:index]]
   142	            right = [item.high for item in bars[index + 1:index + lr + 1]]
   143	            is_swing = all(swing_price > value for value in left + right)
   144	        else:
   145	            raise BacktestContractError(f"invalid side={side}")
   146	        if not is_swing:
   147	            continue
   148	
   149	        confirmed_count += 1
   150	        stop = initial_stop(side, entry, [swing_price], config)
   151	        distance = abs(entry - stop)
   152	        cap = config.initial_swing_max_distance_usd
   153	        if cap > 0 and distance > cap:
   154	            continue
   155	        eligible.append((bar.volume, bar.ts, swing_price, stop))
   156	
   157	    if not eligible:
   158	        reason = "NO_SWING_WITHIN_INITIAL_STOP_CAP" if confirmed_count else "NO_CONFIRMED_VOLUME_SWING"
   159	        raise InitialStopSelectionError(
   160	            reason,
   161	            f"confirmed={confirmed_count} cap={config.initial_swing_max_distance_usd}",
   162	        )
   163	
   164	    swing_volume, swing_ts, swing_price, stop = max(eligible, key=lambda item: (item[0], item[1]))
   165	    return InitialSwingSelection(
   166	        stop=stop,
   167	        swing_ts=swing_ts,
   168	        swing_price=swing_price,
   169	        swing_volume=swing_volume,
   170	        eligible_count=len(eligible),
   171	        confirmed_count=confirmed_count,
   172	    )

   175	def compute_targets(side: str, entry: float, stop: float, config: ReplayConfig) -> tuple[float, float]:
   176	    risk = abs(entry - stop)
   177	    if risk <= 0:
   178	        raise BacktestContractError("initial risk must be positive")
   179	    values: list[float] = []
   180	    for multiplier in config.tp_r_multipliers:
   181	        raw = entry + multiplier * risk if side == "LONG" else entry - multiplier * risk
   182	        values.append(floor_to_step(raw, config.tick_size) if side == "LONG" else ceil_to_step(raw, config.tick_size))
   183	    if len(values) != 2:
   184	        raise BacktestContractError("exactly two TP multipliers are required")
   185	    return values[0], values[1]

   188	def select_candidate_sweep_stop(candidate: Candidate, entry: float,
   189	                                bars: list[FeedBar], config: ReplayConfig) -> InitialSwingSelection:
   190	    """Opt-in research SHORT anchor, verified against the complete as-of feed.
   191	
   192	    Never accepts an arbitrary stop supplied in metadata and never falls back to
   193	    another policy. No production Executor code calls this replay-only function.
   194	    """
   195	    try:
   196	        start = datetime.fromisoformat(candidate.shadow_flags["sweep_start_ts"])
   197	        end = datetime.fromisoformat(candidate.shadow_flags["sweep_end_ts"])
   198	        declared_high = float(candidate.shadow_flags["sweep_high"])
   199	        declared_stop = float(candidate.shadow_flags["sweep_stop_price"])
   200	        buffer = float(candidate.shadow_flags["stop_buffer_fraction"])
   201	    except (KeyError, TypeError, ValueError) as exc:
   202	        raise InitialStopSelectionError("INVALID_CANDIDATE_SWEEP_METADATA", str(exc)) from exc
   203	    if (candidate.side != "SHORT" or start.tzinfo is None or end.tzinfo is None
   204	        or end != candidate.signal_ts_utc or start > end or buffer != .0001
   205	        or not all(math.isfinite(x) for x in (declared_high,declared_stop,buffer))):
   206	        raise InitialStopSelectionError("INVALID_CANDIDATE_SWEEP_INTERVAL")
   207	    window = [b for b in bars if start <= b.ts <= end]
   208	    if (not window or window[0].ts != start or window[-1].ts != end
   209	        or len(window) != int((end-start)/timedelta(minutes=1))+1
   210	        or any(b.is_synthetic or b.volume <= 0 for b in window)
   211	        or any(b.ts-a.ts != timedelta(minutes=1) for a,b in zip(window,window[1:]))):
   212	        raise InitialStopSelectionError("INCOMPLETE_CANDIDATE_SWEEP_INTERVAL")
   213	    highest = max(window, key=lambda b: (b.high,b.ts))
   214	    expected_stop = highest.high*(1+buffer)
   215	    if not math.isclose(declared_high, highest.high, rel_tol=0,abs_tol=1e-8) or not math.isclose(declared_stop, expected_stop, rel_tol=0,abs_tol=1e-8):
   216	        raise InitialStopSelectionError("CANDIDATE_SWEEP_ANCHOR_MISMATCH")
   217	    stop = ceil_to_step(expected_stop, config.tick_size)
   218	    if stop <= entry:
   219	        raise InitialStopSelectionError("CANDIDATE_SWEEP_STOP_NOT_PROTECTIVE")
   220	    return InitialSwingSelection(stop, highest.ts, highest.high, highest.volume, 1, 1)

   223	def build_execution_plan(
   224	    candidate: Candidate,
   225	    reference_bars: list[FeedBar],
   226	    config: ReplayConfig,
   227	    *,
   228	    execution_bars: list[FeedBar] | None = None,
   229	    reference_timestamps: list | None = None,
   230	    execution_timestamps: list | None = None,
   231	) -> ExecutionPlan:
   232	    config.validate()
   233	    execution_bars = execution_bars or reference_bars
   234	    reference_timestamps = reference_timestamps or [bar.ts for bar in reference_bars]
   235	    execution_timestamps = execution_timestamps or [bar.ts for bar in execution_bars]
   236	    cutoff = bisect_right(reference_timestamps, candidate.signal_ts_utc)
   237	    execution_cutoff = bisect_right(execution_timestamps, candidate.signal_ts_utc)
   238	    if cutoff <= 0 or reference_bars[cutoff - 1].ts != candidate.signal_ts_utc:
   239	        raise BacktestContractError("missing exact reference-feed signal minute")
   240	    if execution_cutoff <= 0 or execution_bars[execution_cutoff - 1].ts != candidate.signal_ts_utc:
   241	        raise BacktestContractError("missing exact execution-feed signal minute")
   242	    window_bars = reference_bars[max(0, cutoff - config.swing_lookback_minutes):cutoff]
   243	    if config.initial_stop_policy != "candidate_sweep" and config.initial_swing_require_full_window and len(window_bars) < config.swing_lookback_minutes:
   244	        raise InitialStopSelectionError(
   245	            "NO_FULL_INITIAL_SWING_WINDOW",
   246	            f"bars={len(window_bars)} required={config.swing_lookback_minutes}",
   247	        )
   248	    history_bars = [bar for bar in window_bars if not bar.is_synthetic]
   249	    reference_close = reference_bars[cutoff - 1].close
   250	    execution_close = execution_bars[execution_cutoff - 1].close
   251	    if reference_close <= 0 or execution_close <= 0:
   252	        raise BacktestContractError("invalid conversion reference close")
   253	    if config.price_contour == "btcusdt_signal":
   254	        # Research contour: candidate, plan, barriers, and lifecycle all remain
   255	        # in the signal feed's BTCUSDT quote space. The CLI supplies the same
   256	        # quality-checked bars for reference and execution in this mode.
   257	        execution_close = reference_close
   258	        conversion_ratio = 1.0
   259	    else:
   260	        conversion_ratio = execution_close / reference_close
   261	    if not config.usdt_usdc_ratio_min <= conversion_ratio <= config.usdt_usdc_ratio_max:
   262	        raise BacktestContractError(
   263	            "conversion ratio outside sanity band: "
   264	            f"ratio={conversion_ratio} "
   265	            f"band=[{config.usdt_usdc_ratio_min},{config.usdt_usdc_ratio_max}]"
   266	        )
   267	    entry_usdt = build_entry_price(candidate.side, candidate.signal_price, config)
   268	    selected_swing: InitialSwingSelection | None = None
   269	    if config.initial_stop_policy == "candidate_sweep":
   270	        selected_swing = select_candidate_sweep_stop(candidate, entry_usdt, reference_bars[:cutoff], config)
   271	        stop_usdt = selected_swing.stop
   272	    elif config.initial_stop_policy == "volume_confirmed_swing":
   273	        selected_swing = select_volume_confirmed_initial_stop(
   274	            candidate.side,
   275	            entry_usdt,
   276	            window_bars,
   277	            config,
   278	        )
   279	        stop_usdt = selected_swing.stop
   280	    else:
   281	        if config.initial_swing_price_source == "extreme":
   282	            history = [bar.low if candidate.side == "LONG" else bar.high for bar in history_bars]
   283	        else:
   284	            history = [bar.close for bar in history_bars]
   285	        stop_usdt = initial_stop(candidate.side, entry_usdt, history, config)
   286	    tp1_usdt, tp2_usdt = compute_targets(candidate.side, entry_usdt, stop_usdt, config)
   287	    raw_entry = entry_usdt * conversion_ratio
   288	    raw_stop = stop_usdt * conversion_ratio
   289	    raw_tp1 = tp1_usdt * conversion_ratio
   290	    raw_tp2 = tp2_usdt * conversion_ratio
   291	    signal_close_usdc = candidate.signal_price * conversion_ratio
   292	    if candidate.side == "LONG":
   293	        entry = floor_to_step(raw_entry, config.tick_size)
   294	        entry = max(entry, ceil_to_step(signal_close_usdc + config.tick_size, config.tick_size))
   295	        stop = floor_to_step(raw_stop, config.tick_size)
   296	        tp1 = floor_to_step(raw_tp1, config.tick_size)
   297	        tp2 = floor_to_step(raw_tp2, config.tick_size)
   298	    else:
   299	        entry = ceil_to_step(raw_entry, config.tick_size)
   300	        entry = min(entry, floor_to_step(signal_close_usdc - config.tick_size, config.tick_size))
   301	        stop = ceil_to_step(raw_stop, config.tick_size)
   302	        tp1 = ceil_to_step(raw_tp1, config.tick_size)
   303	        tp2 = ceil_to_step(raw_tp2, config.tick_size)
   304	    if candidate.side == "LONG" and not (stop < entry < tp1 <= tp2):
   305	        raise BacktestContractError("invalid LONG execution plan ordering")
   306	    if candidate.side == "SHORT" and not (stop > entry > tp1 >= tp2):
   307	        raise BacktestContractError("invalid SHORT execution plan ordering")
   308	    return ExecutionPlan(
   309	        candidate_id=candidate.candidate_id,
   310	        side=candidate.side,
   311	        planned_entry_price=entry,
   312	        initial_stop_price=stop,
   313	        initial_risk_usd=abs(entry - stop),
   314	        tp1_price=tp1,
   315	        tp2_price=tp2,
   316	        fixed_notional_usdc=config.fixed_notional_usdc,
   317	        signal_reference_price_usdt=candidate.signal_price,
   318	        reference_feed_close_usdt=reference_close,
   319	        execution_feed_close_usdc=execution_close,
   320	        conversion_ratio=conversion_ratio,
   321	        conversion_reference_ts=candidate.signal_ts_utc,
   322	        planned_entry_price_usdt=entry_usdt,
   323	        initial_stop_price_usdt=stop_usdt,
   324	        tp1_price_usdt=tp1_usdt,
   325	        tp2_price_usdt=tp2_usdt,
   326	        initial_swing_ts=selected_swing.swing_ts if selected_swing else None,
   327	        initial_swing_price_usdt=selected_swing.swing_price if selected_swing else None,
   328	        initial_swing_volume=selected_swing.swing_volume if selected_swing else None,
   329	        initial_swing_eligible_count=selected_swing.eligible_count if selected_swing else 0,
   330	        initial_swing_confirmed_count=selected_swing.confirmed_count if selected_swing else 0,
   331	    )
