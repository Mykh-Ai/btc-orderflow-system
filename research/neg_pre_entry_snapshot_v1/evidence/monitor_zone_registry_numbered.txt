   137	def load_registry(path: str | Path | None) -> pd.DataFrame:
   138	    if path is None:
   139	        return pd.DataFrame(columns=REGISTRY_COLUMNS)
   140	    registry_path = Path(path)
   141	    if not registry_path.exists():
   142	        raise FileNotFoundError(f"Registry input not found: {registry_path}")
   143	    registry = pd.read_csv(registry_path, dtype=str)
   144	    return _normalize_registry(registry)

   147	def build_zone_registry(
   148	    *,
   149	    liquidity_map: pd.DataFrame,
   150	    feed: pd.DataFrame,
   151	    registry_in: pd.DataFrame | None = None,
   152	) -> tuple[pd.DataFrame, dict[str, int]]:
   153	    previous = _normalize_registry(registry_in)
   154	    current_zones = _current_zones_to_registry(liquidity_map, previous)
   155	
   156	    historical = previous[~previous["status"].isin(CARRY_FORWARD_STATUSES)].copy()
   157	    carry = previous[previous["status"].isin(CARRY_FORWARD_STATUSES)].copy()
   158	    candidates = _concat_registry_frames([carry, current_zones])
   159	
   160	    active_rows, merged_rows = _merge_registry_candidates(candidates)
   161	    active_rows = _update_lifecycle(active_rows, feed)
   162	    active_rows = _apply_expiry(active_rows)
   163	
   164	    registry_out = _concat_registry_frames([historical, merged_rows, active_rows])
   165	    registry_out = _normalize_registry(registry_out)
   166	    registry_out = registry_out.sort_values(
   167	        ["first_seen_at", "zone_id", "status"], kind="mergesort"
   168	    ).reset_index(drop=True)
   169	
   170	    stats = {
   171	        "carried_loaded": len(carry),
   172	        "new_created": len(current_zones),
   173	        "carried_forward": int(registry_out["status"].isin(CARRY_FORWARD_STATUSES).sum()),
   174	        "merged": int((registry_out["status"] == "MERGED").sum()),
   175	        "expired": int((registry_out["status"] == "EXPIRED").sum()),
   176	        "crossed_unclassified": int(
   177	            (registry_out["status"] == "CROSSED_UNCLASSIFIED").sum()
   178	        ),
   179	        "active_registry": int((registry_out["status"] == "ACTIVE").sum()),
   180	        "active_forward": int(_active_forward_mask(registry_out).sum()),
   181	        "consumed": int((registry_out["consumption_status"] == "CONSUMED").sum()),
   182	        "chopped_through": int((registry_out["consumption_status"] == "CHOPPED_THROUGH").sum()),
   183	    }
   184	    return registry_out, stats

   375	def _merge_registry_candidates(candidates: pd.DataFrame) -> tuple[pd.DataFrame, pd.DataFrame]:
   376	    candidates = _normalize_registry(candidates)
   377	    if candidates.empty:
   378	        return candidates, pd.DataFrame(columns=REGISTRY_COLUMNS)
   379	    active_rows: list[dict[str, object]] = []
   380	    merged_rows: list[dict[str, object]] = []
   381	
   382	    for side in ["BUY_SIDE", "SELL_SIDE"]:
   383	        side_rows = candidates[candidates["side"] == side].copy()
   384	        side_rows = side_rows.sort_values(
   385	            ["price_lower", "first_seen_at", "zone_id"], kind="mergesort"
   386	        )
   387	        cluster: list[dict[str, object]] = []
   388	        for row in side_rows.to_dict("records"):
   389	            if not cluster:
   390	                cluster = [row]
   391	                continue
   392	            tolerance = _merge_tolerance(cluster + [row])
   393	            current_upper = max(float(item["price_upper"]) for item in cluster)
   394	            if (
   395	                float(row["price_lower"]) <= current_upper + tolerance
   396	                and _registry_merge_sources_compatible(cluster, row)
   397	                and _registry_cluster_width_pct(cluster + [row]) < HARD_WIDE_ZONE_WIDTH_PCT
   398	            ):
   399	                cluster.append(row)
   400	            else:
   401	                active, merged = _collapse_registry_cluster(cluster)
   402	                active_rows.append(active)
   403	                merged_rows.extend(merged)
   404	                cluster = [row]
   405	        if cluster:
   406	            active, merged = _collapse_registry_cluster(cluster)
   407	            active_rows.append(active)
   408	            merged_rows.extend(merged)
   409	    return (
   410	        _normalize_registry(pd.DataFrame(active_rows, columns=REGISTRY_COLUMNS)),
   411	        _normalize_registry(pd.DataFrame(merged_rows, columns=REGISTRY_COLUMNS)),
   412	    )

   415	def _collapse_registry_cluster(
   416	    cluster: list[dict[str, object]]
   417	) -> tuple[dict[str, object], list[dict[str, object]]]:
   418	    if len(cluster) == 1:
   419	        row = cluster[0].copy()
   420	        row["merged_into_zone_id"] = ""
   421	        return row, []
   422	
   423	    target = _merge_target(cluster)
   424	    lower = min(float(row["price_lower"]) for row in cluster)
   425	    upper = max(float(row["price_upper"]) for row in cluster)
   426	    source_level_ids = _pipe_union(row["source_level_ids"] for row in cluster)
   427	    source_timeframes = _pipe_union(row["source_timeframes"] for row in cluster)
   428	    source_zone_types = sorted({str(row["zone_type"]) for row in cluster})
   429	    confidence_components = _merged_confidence_components(
   430	        cluster, source_timeframes, source_zone_types
   431	    )
   432	    confidence_score = int(confidence_components["final_confidence_score"])
   433	    confidence_tier = _confidence_tier(confidence_score)
   434	    instrumentation = score_instrumentation_fields(
   435	        source_level_ids=source_level_ids,
   436	        source_timeframes=source_timeframes,
   437	        zone_type=_registry_merged_zone_type(str(target["side"]), source_zone_types),
   438	        price_lower=lower,
   439	        price_upper=upper,
   440	        price_mid=(lower + upper) / 2,
   441	        confidence_score=confidence_score,
   442	        confidence_tier=confidence_tier,
   443	        score_components=confidence_components,
   444	    )
   445	
   446	    active = target.copy()
   447	    active.update(
   448	        {
   449	            "first_seen_at": min(str(row["first_seen_at"]) for row in cluster),
   450	            "last_seen_at": max(str(row["last_seen_at"]) for row in cluster),
   451	            "last_updated_at": max(str(row["last_updated_at"]) for row in cluster),
   452	            "price_lower": lower,
   453	            "price_upper": upper,
   454	            "price_mid": (lower + upper) / 2,
   455	            "source_level_ids": source_level_ids,
   456	            "source_timeframes": source_timeframes,
   457	            "source_timeframe_primary": _merged_source_timeframe_primary(cluster, source_timeframes),
   458	            "htf_level_type": _merged_htf_text(cluster, "htf_level_type"),

   547	def _update_lifecycle(registry: pd.DataFrame, feed: pd.DataFrame, *, closed_htf_only: bool = False) -> pd.DataFrame:
   548	    registry = _normalize_registry(registry)
   549	    if registry.empty or feed.empty:
   550	        return registry
   551	    frame = feed.sort_values("Timestamp", kind="mergesort")
   552	    run_end = _format_ts(frame["Timestamp"].max())
   553	    updated = []
   554	    for row in registry.to_dict("records"):
   555	        row = row.copy()
   556	        row_first_seen = _format_ts(row["first_seen_at"])
   557	        effective_seen = max(run_end, row_first_seen)
   558	        status, touch_at, cross_at = _path_status(row, frame)
   559	        if touch_at and not row.get("last_touch_at"):
   560	            row["last_touch_at"] = touch_at
   561	        if cross_at:
   562	            if not row.get("last_cross_at") or row["last_cross_at"] != cross_at:
   563	                row["cross_count"] = int(float(row["cross_count"])) + 1
   564	            row["last_cross_at"] = cross_at
   565	        elif touch_at:
   566	            row["touch_count"] = int(float(row["touch_count"])) + 1
   567	            row["last_touch_at"] = touch_at
   568	        row["status"] = status
   569	        row["last_seen_at"] = effective_seen
   570	        row["last_updated_at"] = effective_seen
   571	        row["age_bars"] = _age_bars(row["first_seen_at"], frame)
   572	        row["age_days"] = _age_days(row["first_seen_at"], effective_seen)
   573	        row.update(_consumption_fields(row, frame, closed_htf_only=closed_htf_only))
   574	        if row["consumption_status"] == "REACTED":
   575	            row["status"] = "REACTED"
   576	        if status in CARRY_FORWARD_STATUSES:
   577	            row["active_days"] = max(int(float(row["active_days"])), int(row["age_days"]) + 1)
   578	        row["active_forward_role"] = _active_forward_role(row)
   579	        row["active_forward"] = _bool_text(_is_active_forward(row))
   580	        updated.append(row)
   581	    return _normalize_registry(pd.DataFrame(updated, columns=REGISTRY_COLUMNS))

   584	def _path_status(row: dict[str, object], feed: pd.DataFrame) -> tuple[str, str, str]:
   585	    lower = float(row.get("zone_outer_lower", row["price_lower"]) or row["price_lower"])
   586	    upper = float(row.get("zone_outer_upper", row["price_upper"]) or row["price_upper"])
   587	    after_first_seen = feed[feed["Timestamp"] >= pd.Timestamp(row["first_seen_at"])]
   588	    if after_first_seen.empty:
   589	        return str(row.get("status", "ACTIVE") or "ACTIVE"), "", ""
   590	    if row["side"] == "BUY_SIDE":
   591	        cross = after_first_seen[after_first_seen["HiPrice"] > upper]
   592	        touch = after_first_seen[
   593	            (after_first_seen["HiPrice"] >= lower) & (after_first_seen["LowPrice"] <= upper)
   594	        ]
   595	    else:
   596	        cross = after_first_seen[after_first_seen["LowPrice"] < lower]
   597	        touch = after_first_seen[
   598	            (after_first_seen["LowPrice"] <= upper) & (after_first_seen["HiPrice"] >= lower)
   599	        ]
   600	    touch_at = "" if touch.empty else _format_ts(touch.iloc[0]["Timestamp"])
   601	    cross_at = "" if cross.empty else _format_ts(cross.iloc[0]["Timestamp"])
   602	    if cross_at:
   603	        return "CROSSED_UNCLASSIFIED", touch_at or cross_at, cross_at
   604	    if touch_at:
   605	        return "TOUCHED", touch_at, ""
   606	    return "ACTIVE", "", ""

   651	    core_lower = float(row.get("zone_core_lower", row["price_lower"]) or row["price_lower"])

   661	        "ABOVE" if above else "BELOW" if below else "INSIDE"

   672	    first_sweep_at = previous_first_sweep or (sweep_timestamps[0] if sweep_timestamps else "")

   674	    accepted_above_at = _consecutive_threshold_at(after_first_seen, close_above.tolist())
   675	    accepted_below_at = _consecutive_threshold_at(after_first_seen, close_below.tolist())

   925	def _htf_lifecycle_metrics(
   926	    row: dict[str, object], feed: pd.DataFrame, lower: float, upper: float,
   927	    *, closed_htf_only: bool = False,
   928	) -> dict[str, object]:

   930	    previous_close_through = int(float(row.get("htf_close_through_count", 0) or 0))
   931	    previous_acceptance = int(float(row.get("htf_acceptance_count", 0) or 0))
   932	    history_context_start = str(row.get("history_context_start", "") or row.get("first_seen_at", "") or "")
   933	    if not _has_htf_source(row):
   934	        return {
   935	            "htf_sweep_count": previous_sweeps,
   936	            "htf_close_through_count": previous_close_through,

   939	            "history_context_incomplete": _bool_text(row.get("history_context_incomplete", False)),
   940	        }
   941	
   942	    metric_timeframe = _htf_metric_timeframe(row)
   943	    try:
   944	        htf_frame = aggregate_timeframe(feed, metric_timeframe)
   945	        if closed_htf_only and not htf_frame.empty:
   946	            # Opt-in research adapter: inputs here have minute-OPEN labels.
   947	            # A partial hour, missing minute or duplicate cannot confirm HTF state.
   948	            minutes = {"M15": 15, "H1": 60, "H4": 240}[metric_timeframe]
   949	            rule = {"M15": "15min", "H1": "h", "H4": "4h"}[metric_timeframe]

   952	            complete = (
   953	                htf_frame["Timestamp"].map(counts).eq(minutes)
   954	                & htf_frame["Timestamp"].map(unique).eq(minutes)
   955	                & htf_frame["source_start"].eq(htf_frame["Timestamp"])
   956	                & htf_frame["source_end"].eq(htf_frame["Timestamp"] + pd.Timedelta(minutes=minutes - 1))
   957	            )
   958	            htf_frame = htf_frame.loc[complete].copy()
   959	    except Exception:
   960	        htf_frame = pd.DataFrame()
   961	    if htf_frame.empty:
   962	        return {
   963	            "htf_sweep_count": previous_sweeps,
   964	            "htf_close_through_count": previous_close_through,
   965	            "htf_acceptance_count": previous_acceptance,
   966	            "history_context_start": history_context_start,
   967	            "history_context_incomplete": "true",
   968	        }
   969	
   970	    start_text = str(row.get("htf_confirmation_timestamp", "") or row.get("first_seen_at", ""))
   971	    if start_text:
   972	        htf_frame = htf_frame[htf_frame["Timestamp"] >= pd.Timestamp(start_text)]
   973	    if htf_frame.empty:
   974	        return {
   975	            "htf_sweep_count": previous_sweeps,
   976	            "htf_close_through_count": previous_close_through,
   977	            "htf_acceptance_count": previous_acceptance,
   978	            "history_context_start": history_context_start or _format_ts(feed["Timestamp"].min()),
   979	            "history_context_incomplete": _bool_text(row.get("history_context_incomplete", False)),
   980	        }
   981	
   982	    if str(row.get("side", "")) == "BUY_SIDE":
   983	        swept = pd.to_numeric(htf_frame["HiPrice"], errors="coerce") > upper
   984	        closed_through = pd.to_numeric(htf_frame["ClosePrice"], errors="coerce") > upper
   985	    else:
   986	        swept = pd.to_numeric(htf_frame["LowPrice"], errors="coerce") < lower
   987	        closed_through = pd.to_numeric(htf_frame["ClosePrice"], errors="coerce") < lower
   988	    htf_sweeps = _transition_count(swept.tolist())
   989	    htf_close_through = int(closed_through.sum())
   990	    htf_acceptance = _accepted_run_count(closed_through.tolist())

   992	        "htf_sweep_count": max(previous_sweeps, htf_sweeps),
   993	        "htf_close_through_count": max(previous_close_through, htf_close_through),
   994	        "htf_acceptance_count": max(previous_acceptance, htf_acceptance),
   995	        "history_context_start": history_context_start or _format_ts(htf_frame["source_start"].min()),
   996	        "history_context_incomplete": _bool_text(row.get("history_context_incomplete", False)),

  1035	def _failed_acceptance_count(
  1036	    row: dict[str, object],
  1037	    frame: pd.DataFrame,
  1038	    close_above: list[bool],
  1039	    close_below: list[bool],
  1040	    lower: float,
  1041	    upper: float,
  1042	) -> int:
  1043	    closes = pd.to_numeric(frame["ClosePrice"], errors="coerce").tolist()
  1044	    if str(row.get("side", "")) == "BUY_SIDE":
  1045	        attempt_flags = close_above
  1046	        failed_return = lambda close: float(close) <= upper
  1047	    else:
  1048	        attempt_flags = close_below
  1049	        failed_return = lambda close: float(close) >= lower
  1050	
  1051	    failures = 0
  1052	    idx = 0
  1053	    while idx < len(attempt_flags):
  1054	        if not attempt_flags[idx]:
  1055	            idx += 1
  1056	            continue
  1057	        run_start = idx
  1058	        while idx < len(attempt_flags) and attempt_flags[idx]:
  1059	            idx += 1
  1060	        run_length = idx - run_start
  1061	        if run_length >= ACCEPTANCE_CLOSE_BARS:
  1062	            continue
  1063	        lookahead = closes[idx : idx + ACCEPTANCE_CLOSE_BARS]
  1064	        if any(failed_return(close) for close in lookahead):
  1065	            failures += 1
  1066	    return failures

  1425	def _merged_confidence_components(
  1426	    rows: list[dict[str, object]], source_timeframes: str, source_zone_types: list[str]
  1427	) -> dict[str, object]:
  1428	    prior_score = max(int(float(row["confidence_score"])) for row in rows)
  1429	    bounded_prior_score = min(prior_score, 88)
  1430	    source_count = len(_pipe_union(row["source_level_ids"] for row in rows).split("|"))
  1431	    max_known_source_count = max(
  1432	        len([part for part in str(row["source_level_ids"]).split("|") if part])
  1433	        for row in rows
  1434	    )
  1435	    fresh_source_count = max(source_count - max_known_source_count, 0)
  1436	    source_families = _registry_source_families(source_timeframes, source_zone_types)
  1437	    source_diversity_bonus = min(max(len(source_families) - 1, 0) * 2, 6)
  1438	    fresh_source_bonus = min(fresh_source_count * 3, 6)
  1439	    raw_source_count_bonus = fresh_source_bonus
  1440	    source_count_bonus = fresh_source_bonus
  1441	    h4_component = 2 if "H4" in source_timeframes and fresh_source_count > 0 else 0
  1442	    pdh_pdl_component = (
  1443	        4
  1444	        if any("PDH" in zone_type or "PDL" in zone_type for zone_type in source_zone_types)
  1445	        else 0
  1446	    )
  1447	    width_penalty = _width_penalty(
  1448	        zone_width_pct := _zone_width_pct(
  1449	            min(float(row["price_lower"]) for row in rows),
  1450	            max(float(row["price_upper"]) for row in rows),
  1451	            (
  1452	                min(float(row["price_lower"]) for row in rows)
  1453	                + max(float(row["price_upper"]) for row in rows)
  1454	            )
  1455	            / 2,
  1456	        )
  1457	    )
  1458	    carry_forward_decay_or_penalty = _carry_forward_decay_or_penalty(
  1459	        rows=rows,
  1460	        fresh_source_count=fresh_source_count,
  1461	        source_families=source_families,
  1462	    )
  1463	    data_quality_penalty = (
  1464	        -5 if _quality_values(row["data_quality"] for row in rows) != "RAW" else 0
  1465	    )
  1466	    pre_clamp_score = (
  1467	        bounded_prior_score
  1468	        + source_diversity_bonus
  1469	        + fresh_source_bonus
  1470	        + h4_component
  1471	        + pdh_pdl_component
  1472	        + width_penalty
  1473	        + carry_forward_decay_or_penalty
  1474	        + data_quality_penalty
  1475	    )
  1476	    final_score = max(0, min(100, pre_clamp_score))
  1477	    precision_status = precision_status_for_width(zone_width_pct)
  1478	    return {
  1479	        "base_score": 0,
  1480	        "timeframe_score": h4_component,
  1481	        "timeframe_component_total": h4_component,
  1482	        "h1_component": None,
  1483	        "h4_component": h4_component,
  1484	        "session_component": None,
  1485	        "pdh_pdl_component": pdh_pdl_component,
  1486	        "equal_level_component": None,
  1487	        "source_diversity_bonus": source_diversity_bonus,
  1488	        "raw_source_count_bonus": raw_source_count_bonus,
  1489	        "source_count_bonus": source_count_bonus,
  1490	        "cluster_bonus": 0,
  1491	        "touch_bonus": None,
  1492	        "width_penalty": width_penalty,
  1493	        "precision_status": precision_status,
  1494	        "hard_wide_zone_width_pct": HARD_WIDE_ZONE_WIDTH_PCT,
  1495	        "data_quality_penalty": data_quality_penalty,
  1496	        "carry_forward_decay_or_penalty": carry_forward_decay_or_penalty,
  1497	        "carry_forward_prior_score": prior_score,
  1498	        "bounded_prior_score": bounded_prior_score,
  1499	        "fresh_source_count": fresh_source_count,
  1500	        "pre_clamp_score": pre_clamp_score,
  1501	        "final_confidence_score": final_score,
  1502	        "confidence_tier": _confidence_tier(final_score),
  1503	        "source_level_count": source_count,
  1504	        "cluster_member_count": len(rows),
  1505	        "source_timeframes": source_timeframes,
  1506	        "source_families": "|".join(sorted(source_families)),
  1507	        "source_zone_types": "|".join(source_zone_types),
