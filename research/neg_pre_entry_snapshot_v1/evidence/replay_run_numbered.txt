    17	def stats(frame):
    18	    closed=frame[(frame.entry_status=='FILLED') & frame.lifecycle_class.isin(old.RESOLVED)]
    19	    net=closed.net_pnl_usdc.astype(float); n=len(closed)
    20	    losses=-net[net<0].sum()
    21	    return {'N':n,'wins':int((net>0).sum()),'losses':int((net<0).sum()),'breakeven':int((net==0).sum()),
    22	        'win_rate_pct':float(100*(net>0).sum()/n) if n else None,
    23	        'gross':float(closed.gross_pnl_usdc.sum()),'commission':float(closed.commission_usdc.sum()),
    24	        'slippage':float(closed.slippage_usdc.sum()),'net':float(net.sum()),
    25	        'gross_per_trade':float(closed.gross_pnl_usdc.sum()/n) if n else None,
    26	        'net_per_trade':float(net.sum()/n) if n else None,
    27	        'PF':float(net[net>0].sum()/losses) if losses else None,
    28	        'unresolved':int(((frame.entry_status=='FILLED') & ~frame.lifecycle_class.isin(old.RESOLVED)).sum()),
    29	        'blocked':int((frame.entry_status=='BLOCKED').sum()),'opportunities':len(frame)}

    31	def timing(minutes):
    32	    assert 0<minutes<=240
    33	    return '<=60' if minutes<=60 else '61-120' if minutes<=120 else '121-180' if minutes<=180 else '181-240'
    34	
    35	def signature(c):
    36	    f=c.shadow_flags
    37	    return (f['seed_id'],f['zone_id'],c.signal_ts_utc,c.signal_price,c.delta,c.side,
    38	            f['lower'],f['upper'],f['sweep_start_ts'],f['sweep_high'],f['stop_anchor'])
    39	
    40	def mapping(cs,inventory):
    41	    refs={r['zone_id']:r for r in inventory}
    42	    output=[]
    43	    for c in cs:
    44	        f=c.shadow_flags;ref=refs[f['zone_id']];level=json.loads(ref['level'])
    45	        minutes=(c.signal_ts_utc-datetime.fromisoformat(f['seed_ts'])).total_seconds()/60
    46	        output.append(dict(candidate_id=c.candidate_id,seed_id=f['seed_id'],delta=c.delta,
    47	            seed_ts=f['seed_ts'],confirmation_ts=c.signal_ts_utc.isoformat(),zone_id=f['zone_id'],
    48	            price_lower=f['lower'],price_upper=f['upper'],available_at=f['level_available_at'],
    49	            crossed_at=f['crossed_at'],sweep_start_ts=f['sweep_start_ts'],sweep_high=f['sweep_high'],
    50	            stop_anchor=f['stop_anchor'],significance_score=f['significance_score'],bucket=f['bucket'],
    51	            source_timeframe=ref['timeframe'],level_type=level['level_type'],
    52	            delta_to_confirmation_minutes=minutes,timing_bucket=timing(minutes),
    53	            month=c.signal_ts_utc.strftime('%Y-%m')))
    54	    return pd.DataFrame(output)

    56	def enrich(frame,meta):
    57	    result=frame.merge(meta,on='candidate_id',how='left',validate='one_to_one')
    58	    assert result.level_type.notna().all()
    59	    return result

    61	def summaries(label,mode,frame):
    62	    total=dict(experiment=label,mode=mode,**stats(frame)); rows=[]
    63	    for column,values in [('level_type',['H1_SWING_HIGH','H4_SWING_HIGH']),
    64	                          ('month',sorted(frame.month.unique())),
    65	                          ('timing_bucket',['<=60','61-120','121-180','181-240'])]:
    66	        for value in values:rows.append(dict(experiment=label,mode=mode,dimension=column,cohort=value,**stats(frame[frame[column]==value])))
    67	    # Monthly concentration within each structural type is explicitly auditable.
    68	    for level in ['H1_SWING_HIGH','H4_SWING_HIGH']:
    69	        for month in sorted(frame.month.unique()):
    70	            rows.append(dict(experiment=label,mode=mode,dimension='level_month',cohort=level+'|'+month,
    71	                             **stats(frame[(frame.level_type==level)&(frame.month==month)])))
    72	    return total,rows

    74	def run():
    75	    gate=json.loads((HERE/'baseline_parity/result.json').read_text());assert gate['passed']
    76	    out=HERE/'results';out.mkdir(exist_ok=False)

    85	    old.rows(out/'query_cache_hashes.csv',({'path':str(p.relative_to(old.ROOT)),'sha256':old.digest(p)} for p in cache_files))
    86	    code=old.code_hashes();code.update({str(p.relative_to(old.ROOT)):old.digest(p) for p in sorted(HERE.glob('*.py'))})
    87	    configs={tf:replace(old.CONFIG,experiment_id='NEG_H4_CLOSE_V1' if tf=='H4' else 'NEG_H1_CONTROL',
    88	                description='Numeric NEG only; frozen upper zone; completed canonical '+tf+' close; horizon four hours') for tf in ['H4','H1']}
    89	    for cfg in configs.values():
    90	        assert {k:v for k,v in cfg.to_dict().items() if k not in ['experiment_id','description']}=={
    91	            k:v for k,v in old.CONFIG.to_dict().items() if k not in ['experiment_id','description']}
    92	    old.dump(out/'config.json',{tf:cfg.to_dict() for tf,cfg in configs.items()})
    93	    old.dump(out/'protocol_freeze.json',dict(base_sha='966dabcc0bef0ac7289905fcdfaf93252e1d1198',
    94	        protocol_sha256=old.digest(HERE/'PROTOCOL.md'),code_hashes=code,decision_start=manifest['decision_start'],
    95	        decision_end=manifest['decision_end'],outcome_end=manifest['outcome_end'],
    96	        seed_filter='numeric delta < 0',horizon_minutes=240,confirmation='canonical completed H4 CLOSE < frozen lower',
    97	        baseline_parity_sha256=old.digest(HERE/'baseline_parity/result.json'),
    98	        input_manifest_sha256=old.digest(out/'input_hashes.csv'),cache_manifest_sha256=old.digest(out/'query_cache_hashes.csv'),
    99	        monitor_head=old.head(old.MONITOR),no_fresh_holdout=True,no_production_change=True))
   100	    totals=[];breakdowns=[];funnels={};frames={};all_cs={};checks=[]
   101	    for tf in ['H1','H4']:
   102	        label=configs[tf].experiment_id;folder=out/label;folder.mkdir()
   103	        cache.last_query=None;e=NegCloseEngine(cache,tf)
   104	        old_control=DeltaH1Engine(cache) if tf=='H1' else None
   105	        # Same cache shared for identical timestamps, no query recomputation.
   106	        for b in bars:
   107	            triggers=batches.get(b.ts,())
   108	            e.step(b,triggers)
   109	            if old_control is not None:old_control.step(b,triggers)
   110	        audit=e.finish()
   111	        if old_control is not None:
   112	            old_control.finish()
   113	            assert [signature(c) for c in e.candidates]==[signature(c) for c in old_control.candidates]
   114	            assert [(r['seed_id'],r['status']) for r in audit]==[(r['seed_id'],r['status']) for r in old_control.audit]
   115	            checks.append('NEG-only H1 exact signal/stop/status parity with unchanged original H1 engine')
   116	        seen={r['seed_id'] for r in audit}
   117	        audit.extend(dict(seed_id=s.seed_id,seed_ts=s.ts.isoformat(),delta=s.delta,event=s.event,
   118	            status='NO_SEED_BAR',source_path=s.source_path,source_line=s.source_line) for s in neg if s.seed_id not in seen)
   119	        cs=[c for c in e.candidates if old.replay_eligible(c,manifest)];all_cs[label]=cs
   120	        old.rows(folder/'seed_audit.csv',audit);old.rows(folder/'candidate_events.csv',e.events)

   122	        old.rows(folder/'zone_selection_audit.csv',e.zone_evidence)

   129	        old.rows(folder/'portfolio_events.csv',(r.to_dict() for r in pevents))

   135	            closed=frame[frame.entry_status=='FILLED']
   136	            assert ((closed.gross_pnl_usdc-closed.commission_usdc-closed.slippage_usdc-closed.net_pnl_usdc).abs()<1e-8).all()
   137	            assert (closed.side=='SHORT').all() and (closed.delta<0).all()
   138	            assert (pd.to_datetime(closed.entry_fill_ts,utc=True)>pd.to_datetime(closed.confirmation_ts,utc=True)).all()

   142	        counts=dict(Counter(r['status'] for r in audit));pc=stats(frames[(label,'portfolio')])

   144	            no_eligible_zone=counts.get('NO_ELIGIBLE_ZONE',0),duplicate_pending=counts.get('DUPLICATE_PENDING_ZONE',0),
   145	            expired=counts.get('EXPIRED',0),confirmed_all=len(e.candidates),confirmed_decision_eligible=len(cs),
   146	            independent_trades=stats(frames[(label,'independent')])['N'],portfolio_opened=pc['N'],portfolio_blocked=pc['blocked'],
   147	            statuses=counts)
   148	        assert sum(counts.values())==len(neg)
   149	        for c in cs:
   150	            f=c.shadow_flags;t=c.signal_ts_utc;seedts=datetime.fromisoformat(f['seed_ts'])
   151	            assert c.signal_price<f['price_lower'] and seedts<t<=seedts+old.timedelta(hours=4)
   152	            assert f['price']>f['price_upper'] and datetime.fromisoformat(f['available_at'])<seedts

   157	    # Historical NEG contribution, unchanged all-sign ownership/concurrency.
   158	    raw=json.loads((baseline.PRIOR/'candidates.json').read_text())
   159	    cs=[old.Candidate(**dict(c,signal_ts_utc=datetime.fromisoformat(c['signal_ts_utc']))) for c in raw]
   160	    cs=[c for c in cs if c.delta<0 and old.replay_eligible(c,manifest)]
   161	    meta=mapping(cs,inventory)
   162	    for mode in ['independent','portfolio']:

   168	    old.dump(out/'aggregate_metrics.json',totals);old.rows(out/'aggregate_metrics.csv',totals)
