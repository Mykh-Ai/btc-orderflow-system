    96	def run_significant_zone_selector(
    97	    *,
    98	    input_root: str | Path,
    99	    output_dir: str | Path,
   100	    start: str | date,
   101	    end: str | date,
   102	    max_visible_zones: int = 7,
   103	) -> SignificantZoneSelectorResult:
   104	    start_date = _parse_date(start, "start")
   105	    end_date = _parse_date(end, "end")
   106	    if start_date > end_date:
   107	        raise SignificantZoneSelectorError("start must be <= end")
   108	    if max_visible_zones < 1:
   109	        raise SignificantZoneSelectorError("max_visible_zones must be >= 1")
   110	
   111	    root = Path(input_root)
   112	    out_dir = Path(output_dir)
   113	    if not root.exists() or not root.is_dir():
   114	        raise SignificantZoneSelectorError(f"input root not found: {root}")
   115	    out_dir.mkdir(parents=True, exist_ok=True)
   116	
   117	    loaded = _load_window(root, start_date, end_date)
   118	    liquidity_map = loaded["liquidity_map.csv"]
   119	    if liquidity_map.empty:
   120	        selected = pd.DataFrame(columns=SELECTED_ZONE_COLUMNS)
   121	    else:
   122	        selected = _score_candidates(loaded)
   123	        selected = _assign_visibility(selected, max_visible_zones=max_visible_zones)
   124	        selected = selected.sort_values(
   125	            ["visible_on_snapshot", "significance_score", "source_day", "zone_id"],
   126	            ascending=[False, False, True, True],
   127	            kind="mergesort",
   128	        ).reset_index(drop=True)
   129	        selected["rank"] = range(1, len(selected) + 1)
   130	        selected = selected[SELECTED_ZONE_COLUMNS]
   131	
   132	    selected_path = out_dir / SELECTED_ZONES_CSV
   133	    summary_path = out_dir / SELECTOR_SUMMARY_MD
   134	    manifest_path = out_dir / SELECTOR_MANIFEST_JSON
   135	    selected.to_csv(selected_path, index=False)
   136	
   137	    row_counts = {name: int(len(frame)) for name, frame in loaded.items()}
   138	    missing_flags = _missing_data_flags(loaded)
   139	    summary_path.write_text(
   140	        _render_summary(
   141	            selected=selected,
   142	            row_counts=row_counts,
   143	            input_root=root,
   144	            start_date=start_date,
   145	            end_date=end_date,
   146	            max_visible_zones=max_visible_zones,
   147	            missing_flags=missing_flags,
   148	        ),
   149	        encoding="utf-8",
   150	    )
   151	    manifest = {
   152	        "selector_version": SELECTOR_VERSION,
   153	        "start_date": start_date.isoformat(),
   154	        "end_date": end_date.isoformat(),
   155	        "input_root": str(root),
   156	        "outputs": {
   157	            "selected_zones_csv": str(selected_path),
   158	            "summary_md": str(summary_path),
   159	            "manifest_json": str(manifest_path),
   160	        },
   161	        "max_visible_zones": max_visible_zones,
   162	        "visible_zone_count": _visible_zone_count(selected),
   163	        "visible_side_counts": _visible_side_counts(selected),
   164	        "side_balance_policy": _side_balance_policy(max_visible_zones),
   165	        "total_candidate_count": int(len(selected)),
   166	        "missing_data_flags": missing_flags,
   167	        "repo_commit": _repo_commit(),
   168	        "scope": "research_monitor_selection_only_not_trading_advice",
   169	    }
   170	    manifest_path.write_text(json.dumps(manifest, indent=2, sort_keys=True), encoding="utf-8")
   171	
   172	    visible = selected[selected["visible_on_snapshot"].astype(str) == "true"]
   173	    return SignificantZoneSelectorResult(
   174	        output_dir=out_dir,
   175	        selected_zones_path=selected_path,
   176	        summary_path=summary_path,
   177	        manifest_path=manifest_path,
   178	        total_candidate_count=int(len(selected)),
   179	        visible_zone_count=int(len(visible)),
   180	        visible_buy_side_count=int((visible["side"] == "BUY_SIDE").sum()) if not visible.empty else 0,
   181	        visible_sell_side_count=int((visible["side"] == "SELL_SIDE").sum()) if not visible.empty else 0,
   182	    )

   185	def _load_window(root: Path, start_date: date, end_date: date) -> dict[str, pd.DataFrame]:
   186	    frames: dict[str, list[pd.DataFrame]] = {name: [] for name in INPUT_TABLES}
   187	    for day in _date_range(start_date, end_date):
   188	        day_dir = root / day.isoformat()
   189	        if not day_dir.exists():
   190	            raise SignificantZoneSelectorError(f"missing daily output directory: {day_dir}")
   191	        for filename in INPUT_TABLES:
   192	            path = day_dir / filename
   193	            if path.exists():
   194	                try:
   195	                    frame = pd.read_csv(path)
   196	                except pd.errors.EmptyDataError:
   197	                    frame = pd.DataFrame()
   198	            else:
   199	                frame = pd.DataFrame()
   200	            frame["_source_day"] = day.isoformat()
   201	            frame["_source_path"] = str(path)
   202	            frames[filename].append(frame)
   203	    return {
   204	        name: pd.concat(parts, ignore_index=True, sort=False) if parts else pd.DataFrame()
   205	        for name, parts in frames.items()
   206	    }

   209	def _score_candidates(loaded: dict[str, pd.DataFrame]) -> pd.DataFrame:
   210	    liquidity_map = loaded["liquidity_map.csv"].copy()
   211	    event_log = loaded["event_log.csv"]
   212	    observations = loaded["post_sweep_observation.csv"]
   213	    event_groups = _group_by_day_zone(event_log)
   214	    observation_groups = _group_by_day_zone(observations)
   215	
   216	    rows = []
   217	    for _, zone in liquidity_map.iterrows():
   218	        source_day = _string(zone.get("_source_day"))
   219	        zone_id = _string(zone.get("zone_id"))
   220	        events = event_groups.get((source_day, zone_id), pd.DataFrame())
   221	        zone_observations = observation_groups.get((source_day, zone_id), pd.DataFrame())
   222	        rows.append(_score_zone(zone, events, zone_observations))
   223	    frame = pd.DataFrame(rows)
   224	    return frame.sort_values(
   225	        ["significance_score", "source_day", "zone_id"],
   226	        ascending=[False, True, True],
   227	        kind="mergesort",
   228	    ).reset_index(drop=True)

   231	def _score_zone(zone: pd.Series, events: pd.DataFrame, observations: pd.DataFrame) -> dict[str, object]:
   232	    score_components: dict[str, float] = {}
   233	    penalty_components: dict[str, float] = {}
   234	    evidence_present: set[str] = set()
   235	    evidence_missing: set[str] = set()
   236	    reasons: list[str] = []
   237	
   238	    source_day = _string(zone.get("_source_day"))
   239	    zone_id = _string(zone.get("zone_id"))
   240	    side = _string(zone.get("side")) or "UNKNOWN"
   241	    source_timeframe = _string(zone.get("source_timeframes")) or "UNKNOWN"
   242	    source_family = _source_family(source_timeframe)
   243	    label = _first_nonempty(
   244	        zone.get("zone_type"),
   245	        zone.get("htf_level_type"),
   246	        _score_level_types(zone.get("score_components_json")),
   247	    )
   248	    level_type = _first_nonempty(zone.get("htf_level_type"), _score_level_types(zone.get("score_components_json")))
   249	    status = _string(zone.get("status")) or "UNKNOWN"
   250	    lifecycle_status = _first_nonempty(zone.get("htf_lifecycle_status"), zone.get("consumption_status"), status)
   251	    confidence = _string(zone.get("confidence_tier")) or "UNKNOWN"
   252	    representative_price = _float(zone.get("price_mid"))
   253	    distance = _float(zone.get("distance_from_close_pct"))
   254	    current_price = _current_price(representative_price, distance)
   255	    sources = _split_sources(source_timeframe)
   256	
   257	    _add_source_scores(zone, sources, score_components, evidence_present, reasons)
   258	    _add_lifecycle_scores(status, lifecycle_status, score_components, penalty_components, reasons)
   259	    _add_confidence_scores(confidence, score_components, penalty_components, reasons)
   260	    _add_distance_scores(distance, score_components, penalty_components, evidence_present, reasons)
   261	    _add_interaction_scores(zone, score_components, evidence_present, reasons)
   262	    _add_precision_scores(zone, score_components, penalty_components, reasons)
   263	    _add_event_scores(events, score_components, penalty_components, evidence_present, evidence_missing, reasons)
   264	    _add_observation_scores(
   265	        observations,
   266	        score_components,
   267	        evidence_present,
   268	        evidence_missing,
   269	        reasons,
   270	    )
   271	    _add_context_penalties(sources, events, observations, penalty_components, reasons)
   272	
   273	    missing_data = _zone_missing_data_notes()
   274	    evidence_missing.update(missing_data)
   275	
   276	    raw_score = sum(score_components.values()) + sum(penalty_components.values())
   277	    bucket = _bucket_for_zone(raw_score, sources, zone, status, events, observations)
   278	    flow_summary = _flow_summary(events, observations)
   279	    hidden_reason = _base_hidden_reason(
   280	        bucket=bucket,
   281	        status=status,
   282	        lifecycle_status=lifecycle_status,
   283	        events=events,
   284	        observations=observations,
   285	        sources=sources,
   286	    )
   287	
   288	    return {
   289	        "rank": 0,
   290	        "zone_id": zone_id,
   291	        "source_day": source_day,
   292	        "side": side,
   293	        "source_timeframe": source_timeframe,
   294	        "source_family": source_family,
   295	        "level_type": level_type,
   296	        "label": label,
   297	        "price_lower": zone.get("price_lower", ""),
   298	        "price_upper": zone.get("price_upper", ""),
   299	        "representative_price": representative_price,
   300	        "current_price": current_price,
   301	        "distance_to_current_price_pct": distance,
   302	        "lifecycle_status": lifecycle_status,
   303	        "status": status,
   304	        "confidence": confidence,
   305	        "bucket": bucket,
   306	        "significance_score": round(float(raw_score), 3),
   307	        "score_components_json": json.dumps(score_components, sort_keys=True),
   308	        "penalty_components_json": json.dumps(penalty_components, sort_keys=True),
   309	        "evidence_fields_present": "|".join(sorted(evidence_present)),
   310	        "evidence_fields_missing": "|".join(sorted(evidence_missing)),
   311	        "flow_evidence_summary": flow_summary,
   312	        "reason_selected": "; ".join(reasons[:12]),
   313	        "reason_hidden": hidden_reason,
   314	        "visible_on_snapshot": "false",
   315	    }

   318	def _assign_visibility(frame: pd.DataFrame, *, max_visible_zones: int) -> pd.DataFrame:
   319	    out = frame.copy()
   320	    out["visible_on_snapshot"] = "false"
   321	    eligible = out[out.apply(_is_visibility_eligible, axis=1)].copy()
   322	    if eligible.empty:
   323	        return out
   324	
   325	    eligible_indices = _visibility_sorted_indices(out, eligible.index)
   326	    selected_indices = eligible_indices[:max_visible_zones]
   327	    selected_indices = _rebalance_visible_sides(
   328	        out,
   329	        eligible_indices=eligible_indices,
   330	        selected_indices=selected_indices,
   331	        max_visible_zones=max_visible_zones,
   332	    )
   333	
   334	    out.loc[selected_indices, "visible_on_snapshot"] = "true"
   335	    out.loc[out["visible_on_snapshot"] == "true", "reason_hidden"] = ""
   336	    out.loc[
   337	        (out["visible_on_snapshot"] == "false") & (out["reason_hidden"] == ""),
   338	        "reason_hidden",
   339	    ] = "outside max visible zone cap or lower ranked than selected zones"
   340	    return out

   343	def _rebalance_visible_sides(
   344	    frame: pd.DataFrame,
   345	    *,
   346	    eligible_indices: list[int],
   347	    selected_indices: list[int],
   348	    max_visible_zones: int,
   349	) -> list[int]:
   350	    if max_visible_zones < 2:
   351	        return selected_indices
   352	
   353	    selected = list(selected_indices)
   354	    available_sides = {
   355	        _string(frame.loc[idx, "side"])
   356	        for idx in eligible_indices
   357	        if _string(frame.loc[idx, "side"]) in {"BUY_SIDE", "SELL_SIDE"}
   358	    }
   359	    if not {"BUY_SIDE", "SELL_SIDE"}.issubset(available_sides):
   360	        return selected
   361	
   362	    selected = _promote_to_side_minimum(
   363	        frame,
   364	        eligible_indices=eligible_indices,
   365	        selected_indices=selected,
   366	        side="BUY_SIDE",
   367	        target_count=1,
   368	        require_visible_side=True,
   369	    )
   370	    selected = _promote_to_side_minimum(
   371	        frame,
   372	        eligible_indices=eligible_indices,
   373	        selected_indices=selected,
   374	        side="SELL_SIDE",
   375	        target_count=1,
   376	        require_visible_side=True,
   377	    )
   378	
   379	    if max_visible_zones >= 6:
   380	        for side in ["BUY_SIDE", "SELL_SIDE"]:
   381	            if _eligible_side_count(frame, eligible_indices, side) >= 2:
   382	                selected = _promote_to_side_minimum(
   383	                    frame,
   384	                    eligible_indices=eligible_indices,
   385	                    selected_indices=selected,
   386	                    side=side,
   387	                    target_count=2,
   388	                    require_visible_side=False,
   389	                )
   390	    return _visibility_sorted_indices(frame, selected)[:max_visible_zones]

   393	def _promote_to_side_minimum(
   394	    frame: pd.DataFrame,
   395	    *,
   396	    eligible_indices: list[int],
   397	    selected_indices: list[int],
   398	    side: str,
   399	    target_count: int,
   400	    require_visible_side: bool,
   401	) -> list[int]:
   402	    selected = list(selected_indices)
   403	    while _selected_side_count(frame, selected, side) < target_count:
   404	        candidate = _best_hidden_side_candidate(frame, eligible_indices, selected, side)
   405	        if candidate is None:
   406	            return selected
   407	        displaced = _replacement_index_for_side_balance(
   408	            frame,
   409	            selected,
   410	            candidate,
   411	            require_visible_side=require_visible_side,
   412	        )
   413	        if displaced is None:
   414	            return selected
   415	        selected.remove(displaced)
   416	        selected.append(candidate)
   417	        _append_reason(frame, candidate, "reason_selected", SIDE_BALANCE_PROMOTION_REASON)
   418	        frame.loc[displaced, "reason_hidden"] = SIDE_BALANCE_DISPLACED_REASON
   419	    return selected

   422	def _best_hidden_side_candidate(
   423	    frame: pd.DataFrame,
   424	    eligible_indices: list[int],
   425	    selected_indices: list[int],
   426	    side: str,
   427	) -> int | None:
   428	    selected = set(selected_indices)
   429	    for idx in eligible_indices:
   430	        if idx in selected:
   431	            continue
   432	        row = frame.loc[idx]
   433	        if _string(row.get("side")) != side:
   434	            continue
   435	        if _is_side_balance_candidate(row):
   436	            return idx
   437	    return None

   440	def _replacement_index_for_side_balance(
   441	    frame: pd.DataFrame,
   442	    selected_indices: list[int],
   443	    candidate_idx: int,
   444	    *,
   445	    require_visible_side: bool,
   446	) -> int | None:
   447	    candidate = frame.loc[candidate_idx]
   448	    candidate_side = _string(candidate.get("side"))
   449	    candidate_score = _float(candidate.get("significance_score"))
   450	    side_counts = {
   451	        side: _selected_side_count(frame, selected_indices, side)
   452	        for side in ["BUY_SIDE", "SELL_SIDE"]
   453	    }
   454	    overrepresented_sides = [
   455	        side for side, count in side_counts.items() if side != candidate_side and count > 1
   456	    ]
   457	    if not overrepresented_sides:
   458	        return None
   459	
   460	    replacement_pool = [
   461	        idx
   462	        for idx in selected_indices
   463	        if _string(frame.loc[idx].get("side")) in overrepresented_sides
   464	    ]
   465	    for replace_idx in reversed(_visibility_sorted_indices(frame, replacement_pool)):
   466	        replacement = frame.loc[replace_idx]
   467	        if require_visible_side:
   468	            return replace_idx
   469	        replacement_score = _float(replacement.get("significance_score"))
   470	        if replacement_score - candidate_score > 10:
   471	            continue
   472	        if _candidate_can_replace(candidate, replacement):
   473	            return replace_idx
   474	    return None

   477	def _candidate_can_replace(candidate: pd.Series, replacement: pd.Series) -> bool:
   478	    candidate_distance = abs(_float(candidate.get("distance_to_current_price_pct"), 999.0))
   479	    replacement_distance = abs(_float(replacement.get("distance_to_current_price_pct"), 999.0))
   480	    if candidate_distance < replacement_distance:
   481	        return True
   482	    if _bucket_strength(candidate.get("bucket")) > _bucket_strength(replacement.get("bucket")):
   483	        return True
   484	    return _source_strength(candidate.get("source_timeframe")) > _source_strength(replacement.get("source_timeframe"))
   485	
   486	
   487	def _is_side_balance_candidate(row: pd.Series) -> bool:
   488	    if _string(row.get("bucket")) not in {BUCKET_MAJOR, BUCKET_PRIMARY}:
   489	        return False
   490	    if not _valid_price_range(row):
   491	        return False
   492	    if _string(row.get("side")) not in {"BUY_SIDE", "SELL_SIDE"}:
   493	        return False
   494	    distance = abs(_float(row.get("distance_to_current_price_pct"), 999.0))
   495	    return distance <= 2.0 or _has_strong_structural_or_event_evidence(row)
   496	
   497	
   498	def _valid_price_range(row: pd.Series) -> bool:
   499	    lower = _float(row.get("price_lower"))
   500	    upper = _float(row.get("price_upper"))
   501	    return not pd.isna(lower) and not pd.isna(upper) and lower <= upper
   502	
   503	
   504	def _has_strong_structural_or_event_evidence(row: pd.Series) -> bool:
   505	    present = set(_string(row.get("evidence_fields_present")).split("|"))
   506	    return bool({"has_h4_source", "has_h1_source", "event_log", "post_sweep_observation"} & present)

   509	def _eligible_side_count(frame: pd.DataFrame, eligible_indices: list[int], side: str) -> int:
   510	    return sum(1 for idx in eligible_indices if _string(frame.loc[idx].get("side")) == side)
   511

   513	def _selected_side_count(frame: pd.DataFrame, selected_indices: list[int], side: str) -> int:
   514	    return sum(1 for idx in selected_indices if _string(frame.loc[idx].get("side")) == side)

   516	
   517	def _visibility_sorted_indices(frame: pd.DataFrame, indices: Iterable[int]) -> list[int]:
   518	    return sorted((int(idx) for idx in indices), key=lambda idx: _visibility_priority_key(frame.loc[idx], idx))
   519	
   520	
   521	def _visibility_priority_key(row: pd.Series, idx: int) -> tuple[float, float, int, int, str, int]:
   522	    return (
   523	        -_float(row.get("significance_score")),
   524	        abs(_float(row.get("distance_to_current_price_pct"), 999.0)),
   525	        -_bucket_strength(row.get("bucket")),
   526	        -_source_strength(row.get("source_timeframe")),
   527	        _string(row.get("zone_id")),
   528	        idx,
   529	    )

   532	def _bucket_strength(bucket: object) -> int:
   533	    return {

   593	def _side_balance_policy(max_visible_zones: int) -> str:
   594	    if max_visible_zones >= 6:
   595	        return "require one zone per side when available; prefer two zones per side from eligible MAJOR/PRIMARY zones within score gap"

   601	def _side_balance_note(selected: pd.DataFrame, max_visible_zones: int) -> str:
   602	    if selected.empty:
   603	        return "no candidate zones"
   604	    visible_counts = _visible_side_counts(selected)
   605	    if max_visible_zones < 6:
   606	        return "visible cap below six only guarantees one zone per side when available"
   607	    notes = []
   608	    for side in ["BUY_SIDE", "SELL_SIDE"]:
   609	        eligible_count = int(
   610	            selected[
   611	                (selected["side"] == side)
   612	                & selected.apply(_is_visibility_eligible, axis=1)
   613	                & selected.apply(_is_side_balance_candidate, axis=1)
   614	            ].shape[0]
   615	        )
   616	        visible_count = visible_counts.get(side, 0)
   617	        if eligible_count < 2:
   618	            notes.append(f"{side} has fewer than two high-quality eligible zones")
   619	        elif visible_count < 2:
   620	            notes.append(f"{side} has fewer than two visible zones because score-gap replacement criteria were not met")
   621	    return "; ".join(notes) if notes else "both sides preserved when high-quality eligible zones were available"

   856	    score: float,
   857	    sources: set[str],
   858	    zone: pd.Series,
   859	    status: str,
   860	    events: pd.DataFrame,
   861	    observations: pd.DataFrame,
   862	) -> str:
   863	    if status in HIDDEN_STATUSES:
   864	        return BUCKET_NOISE
   865	    if sources.issubset({"SESSION"}) and events.empty and observations.empty:
   866	        return BUCKET_LOCAL_CONTEXT if score >= 35 else BUCKET_NOISE
   867	    has_major = "H4" in sources or _truth(zone.get("has_h4_source")) or _truth(zone.get("has_pdh_pdl_source"))
   868	    if score >= 85 and has_major:

   871	        return BUCKET_PRIMARY

   873	        return BUCKET_MINIMUM
   874	    if score >= 35:
   875	        return BUCKET_LOCAL_CONTEXT
   876	    return BUCKET_NOISE
