     1	"""Additive 39A monitor snapshot contract.
     2	
     3	This module does not replace the existing monitor engine.  It adds the pieces
     4	that were missing at the Executor boundary: exact closed minute windows,
     5	explicit data quality/lineage, and a small persisted state envelope that can
     6	carry the descriptive monitor state across calls and UTC day boundaries.
     7	
     8	The state envelope is intentionally descriptive.  It is not a setup engine,
     9	signal gate, order manager, or outcome recorder.
    10	"""
    11	from __future__ import annotations
    12	
    13	import hashlib
    14	import json
    15	import os
    16	import tempfile
    17	from datetime import date, datetime, timezone
    18	from pathlib import Path
    19	from typing import Any, Mapping
    20	
    21	import pandas as pd

    23	
    24	SNAPSHOT_SCHEMA_VERSION = "market_monitor_snapshot_v39a"
    25	STATE_SCHEMA_VERSION = "SHI_RESET_39A_RUNTIME_STATE_CARRY_FORWARD_V1"
    26	BASE_CONTRACT_VERSION = "market_monitor_snapshot_v1"
    27	LOCAL_WINDOWS_MINUTES = (5, 15, 30, 60, 240, 1440)
    28	_ALIASES = {
    29	    "Open": "OpenPrice",
    30	    "High": "HiPrice",
    31	    "Low": "LowPrice",
    32	    "Close": "ClosePrice",
    33	    "Volume": "TotalQty",
    34	    "AggTrades": "Trades",
    35	}
    36	_REQUIRED_PRICE = ("OpenPrice", "HiPrice", "LowPrice", "ClosePrice")
    37	_NUMERIC = (
    38	    "OpenPrice",
    39	    "HiPrice",
    40	    "LowPrice",
    41	    "ClosePrice",
    42	    "TotalQty",
    43	    "Trades",
    44	    "BuyQty",
    45	    "SellQty",
    46	    "OpenInterest",
    47	    "FundingRate",
    48	)

    51	class MonitorSnapshotV39AError(ValueError):
    52	    """Raised when the v39A monitor input or state contract is invalid."""

    55	def build_market_monitor_snapshot_v39a(
    56	    feed: pd.DataFrame,
    57	    *,
    58	    context_feed: pd.DataFrame | None = None,
    59	    cutoff_ts: Any | None = None,
    60	    state_path: str | Path | None = None,
    61	    symbol: str = "",
    62	    base_snapshot: Mapping[str, Any] | None = None,
    63	    lineage: Mapping[str, Any] | None = None,
    64	    persist_state: bool = True,
    65	) -> dict[str, Any]:
    66	    """Build an entry-time monitor snapshot from closed feed rows.
    67	
    68	    ``base_snapshot`` may contain the existing v1 structure/zone summaries.
    69	    When supplied, those summaries are carried under an explicit v1 lineage
    70	    field; they are never silently presented as new 39A logic.
    71	    """
    72	    frame = _normalise_feed(feed)
    73	    if frame.empty:
    74	        raise MonitorSnapshotV39AError("feed is empty")
    75	    cutoff = _resolve_cutoff(frame, cutoff_ts)
    76	    current = frame[frame["Timestamp"] <= cutoff].copy()
    77	    if current.empty:
    78	        raise MonitorSnapshotV39AError("no feed rows at or before cutoff")
    79	    context = _normalise_feed(context_feed) if context_feed is not None else current.copy()
    80	    context = context[context["Timestamp"] <= cutoff].copy()
    81	    previous_state, state_load = _load_state(state_path)
    82	    windows = {
    83	        str(minutes): _window_metrics(current, minutes=minutes, cutoff=cutoff)
    84	        for minutes in LOCAL_WINDOWS_MINUTES
    85	    }
    86	    quality = _quality_summary(frame, current, context, windows, cutoff)
    87	    current_price = float(current.iloc[-1]["ClosePrice"])
    88	    structure = _structure_summary(windows, current_price)
    89	    zones = _zone_summary(base_snapshot)
    90	    market_state = _market_state_summary(base_snapshot, windows)
    91	    state_memory, next_state = _advance_state(
    92	        previous_state=previous_state,
    93	        state_load=state_load,
    94	        cutoff=cutoff,
    95	        structure=structure,
    96	        zones=zones,
    97	        market_state=market_state,
    98	        quality=quality,
    99	    )
   100	    effective_zones = zones if zones.get("available") else next_state.get("liquidity_zones", zones)
   101	    if state_path is not None and persist_state:
   102	        _write_state(Path(state_path), next_state)
   103	
   104	    source_files = sorted({str(x) for x in current["SourceFile"].dropna().tolist()})
   105	    snapshot = {
   106	        "schema_version": SNAPSHOT_SCHEMA_VERSION,
   107	        "cutoff_ts": _format_ts(cutoff),
   108	        "symbol": symbol,
   109	        "bar_contract": {
   110	            "timestamp_semantics": "UTC_minute_close_label",
   111	            "closed_rows_only": True,
   112	            "cutoff_inclusive": True,
   113	            "windows_minutes": list(LOCAL_WINDOWS_MINUTES),
   114	        },
   115	        "current": {
   116	            "price": round(current_price, 8),
   117	            "rows_used": int(len(current)),
   118	            "last_timestamp": _format_ts(current["Timestamp"].iloc[-1]),
   119	        },
   120	        "windows": windows,
   121	        "market_state": market_state,
   122	        "structure": structure,
   123	        "liquidity_zones": effective_zones,
   124	        "state_memory": state_memory,
   125	        "quality": quality,
   126	        "lineage": {
   127	            "source_files": source_files,
   128	            "source_file_count": len(source_files),
   129	            "source_hashes": dict((lineage or {}).get("source_hashes", {})),
   130	            "feed_identity": (lineage or {}).get("feed_identity", "") or "",
   131	            "base_snapshot_schema": (
   132	                str((base_snapshot or {}).get("schema_version", ""))
   133	                if base_snapshot is not None
   134	                else ""
   135	            ),
   136	            "base_monitor_contract": BASE_CONTRACT_VERSION if base_snapshot is not None else "not_supplied",
   137	            "state_file": str(state_path) if state_path is not None else "",
   138	            "state_schema_version": STATE_SCHEMA_VERSION,
   139	        },
   140	        "boundary": "descriptive monitor state only; no orders, outcomes, PnL, or trailing",
   141	    }
   142	    return _json_safe(snapshot)

   145	def _normalise_feed(feed: pd.DataFrame | None) -> pd.DataFrame:
   146	    if feed is None:
   147	        return pd.DataFrame()
   148	    frame = feed.copy()
   149	    frame = frame.rename(columns={key: value for key, value in _ALIASES.items() if key in frame.columns})
   150	    if "Timestamp" not in frame.columns:
   151	        raise MonitorSnapshotV39AError("feed missing Timestamp")
   152	    missing = [column for column in _REQUIRED_PRICE if column not in frame.columns]
   153	    if missing:
   154	        raise MonitorSnapshotV39AError("feed missing price columns: " + ", ".join(missing))
   155	    for column in _NUMERIC:
   156	        if column not in frame.columns:
   157	            frame[column] = 0.0
   158	        frame[column] = pd.to_numeric(frame[column], errors="coerce")
   159	        if frame[column].isna().any():
   160	            raise MonitorSnapshotV39AError(f"invalid numeric values in {column}")
   161	    frame["Timestamp"] = pd.to_datetime(frame["Timestamp"], utc=True, errors="coerce")
   162	    if frame["Timestamp"].isna().any():
   163	        raise MonitorSnapshotV39AError("invalid Timestamp values")
   164	    if "DataQuality" not in frame.columns:
   165	        if "IsSynthetic" in frame.columns:
   166	            synthetic = pd.to_numeric(frame["IsSynthetic"], errors="coerce").fillna(0)
   167	            frame["DataQuality"] = synthetic.ne(0).map({True: "RECOVERED_DEGRADED", False: "RAW"})
   168	        else:
   169	            frame["DataQuality"] = "RAW"
   170	    frame["DataQuality"] = frame["DataQuality"].fillna("UNKNOWN").astype(str)
   171	    if "SourceFile" not in frame.columns:
   172	        frame["SourceFile"] = "in_memory_feed"
   173	    return frame.sort_values("Timestamp", kind="mergesort").reset_index(drop=True)

   176	def _resolve_cutoff(frame: pd.DataFrame, cutoff_ts: Any | None) -> pd.Timestamp:
   177	    cutoff = pd.Timestamp(cutoff_ts) if cutoff_ts is not None else pd.Timestamp(frame["Timestamp"].max())
   178	    if cutoff.tzinfo is None:
   179	        cutoff = cutoff.tz_localize("UTC")
   180	    else:
   181	        cutoff = cutoff.tz_convert("UTC")
   182	    cutoff = cutoff.floor("min")
   183	    if cutoff > frame["Timestamp"].max():
   184	        raise MonitorSnapshotV39AError("cutoff is later than the available feed")
   185	    return cutoff

   188	def _window_metrics(frame: pd.DataFrame, *, minutes: int, cutoff: pd.Timestamp) -> dict[str, Any]:
   189	    start = cutoff - pd.Timedelta(minutes=minutes - 1)
   190	    window = frame[(frame["Timestamp"] >= start) & (frame["Timestamp"] <= cutoff)].copy()
   191	    expected = pd.date_range(start, cutoff, freq="min", tz="UTC")
   192	    actual = pd.DatetimeIndex(window["Timestamp"])
   193	    missing = expected.difference(actual)
   194	    duplicate_count = int(actual.duplicated().sum())
   195	    complete = len(missing) == 0 and len(window) == len(expected) and duplicate_count == 0
   196	    quality_counts = {
   197	        str(key): int(value)
   198	        for key, value in window["DataQuality"].value_counts().sort_index().items()
   199	    }
   200	    recovered = any(key == "RECOVERED_DEGRADED" for key in quality_counts)
   201	    status = "COMPLETE_RECOVERED_DEGRADED" if complete and recovered else "COMPLETE_RAW" if complete else "PARTIAL"
   202	    if window.empty:
   203	        return {
   204	            "start_timestamp": _format_ts(start),
   205	            "end_timestamp": _format_ts(cutoff),
   206	            "expected_rows": int(len(expected)),
   207	            "rows_used": 0,
   208	            "complete": False,
   209	            "status": "EMPTY",
   210	            "missing_timestamps": [_format_ts(ts) for ts in expected[:10]],
   211	            "duplicate_rows": 0,
   212	            "data_quality_counts": {},
   213	        }
   214	    total_qty = float(window["TotalQty"].sum())
   215	    open_price = float(window.iloc[0]["OpenPrice"])
   216	    close_price = float(window.iloc[-1]["ClosePrice"])
   217	    return {
   218	        "start_timestamp": _format_ts(start),
   219	        "end_timestamp": _format_ts(cutoff),
   220	        "expected_rows": int(len(expected)),
   221	        "rows_used": int(len(window)),
   222	        "complete": bool(complete),
   223	        "status": status,
   224	        "missing_timestamps": [_format_ts(ts) for ts in missing[:10]],
   225	        "duplicate_rows": duplicate_count,
   226	        "data_quality_counts": quality_counts,
   227	        "recovered_degraded": bool(recovered),
   228	        "open": round(open_price, 8),
   229	        "close": round(close_price, 8),
   230	        "high": round(float(window["HiPrice"].max()), 8),
   231	        "low": round(float(window["LowPrice"].min()), 8),
   232	        "price_change_pct": round((close_price / open_price - 1.0) * 100.0, 8) if open_price else None,
   233	        "total_qty": round(total_qty, 8),
   234	        "delta": round(float((window["BuyQty"] - window["SellQty"]).sum()), 8),
   235	        "delta_pct": round(float((window["BuyQty"] - window["SellQty"]).sum()) / total_qty, 10) if total_qty else 0.0,
   236	        "open_interest_change": round(float(window.iloc[-1]["OpenInterest"] - window.iloc[0]["OpenInterest"]), 8),
   237	        "funding_last": round(float(window.iloc[-1]["FundingRate"]), 12),
   238	    }

   241	def _quality_summary(
   242	    input_frame: pd.DataFrame,
   243	    current: pd.DataFrame,
   244	    context: pd.DataFrame,
   245	    windows: Mapping[str, Mapping[str, Any]],
   246	    cutoff: pd.Timestamp,
   247	) -> dict[str, Any]:
   248	    future_rows = int((input_frame["Timestamp"] > cutoff).sum())
   249	    missing_windows = [key for key, value in windows.items() if not value.get("complete")]
   250	    counts = {
   251	        str(key): int(value)
   252	        for key, value in current["DataQuality"].value_counts().sort_index().items()
   253	    }
   254	    return {
   255	        "cutoff_is_latest_used_row": bool(current["Timestamp"].max() == cutoff),
   256	        "future_rows_excluded": future_rows,
   257	        "current_rows": int(len(current)),
   258	        "context_rows": int(len(context)),
   259	        "current_data_quality_counts": counts,
   260	        "recovered_degraded_present": "RECOVERED_DEGRADED" in counts,
   261	        "incomplete_windows": missing_windows,
   262	        "readiness": "READY_WITH_RECOVERED_DATA" if "RECOVERED_DEGRADED" in counts else "READY",
   263	    }

   266	def _structure_summary(windows: Mapping[str, Mapping[str, Any]], current_price: float) -> dict[str, Any]:
   267	    levels: list[dict[str, Any]] = []
   268	    for key in ("240", "1440"):
   269	        metrics = windows.get(key, {})
   270	        if not metrics.get("rows_used"):
   271	            continue
   272	        for role, field in (("SUPPORT_OBSERVED_LOW", "low"), ("RESISTANCE_OBSERVED_HIGH", "high")):
   273	            levels.append(
   274	                {
   275	                    "level_id": f"{key}m_{field}",
   276	                    "role": role,
   277	                    "price": metrics.get(field),
   278	                    "source_window_minutes": int(key),
   279	                    "observed_at": metrics.get("end_timestamp", ""),
   280	                    "lifecycle_status": "OBSERVED_EXTREME",
   281	                }
   282	            )
   283	    return {
   284	        "current_price": round(current_price, 8),
   285	        "levels": levels,
   286	        "contract": "observed_window_extremes_only; no promoted setup lifecycle",
   287	    }

   290	def _zone_summary(base_snapshot: Mapping[str, Any] | None) -> dict[str, Any]:
   291	    if not base_snapshot:
   292	        return {
   293	            "available": False,
   294	            "source": "not_supplied",
   295	            "significant_market_zones": [],
   296	            "liquidity_zones": [],
   297	        }
   298	    return {
   299	        "available": True,
   300	        "source": "base_monitor_snapshot_v1",
   301	        "significant_market_zones": base_snapshot.get("significant_market_zones", {}),
   302	        "liquidity_zones": base_snapshot.get("liquidity_zones", {}),
   303	    }

   306	def _market_state_summary(base_snapshot: Mapping[str, Any] | None, windows: Mapping[str, Mapping[str, Any]]) -> dict[str, Any]:
   307	    if base_snapshot:
   308	        structure_state = base_snapshot.get("market_structure_state")
   309	        if isinstance(structure_state, Mapping) and structure_state:
   310	            return {"source": "base_monitor_snapshot_v1", "value": _json_safe(structure_state)}
   311	        market_state = base_snapshot.get("market_state")
   312	        if isinstance(market_state, Mapping) and market_state:
   313	            return {"source": "base_monitor_snapshot_v1", "value": _json_safe(market_state)}
   314	    m = windows.get("1440", {})
   315	    delta = float(m.get("delta", 0.0) or 0.0)
   316	    change = float(m.get("price_change_pct", 0.0) or 0.0)
   317	    bias = "BULLISH_FLOW" if delta > 0 and change >= 0 else "BEARISH_FLOW" if delta < 0 and change <= 0 else "MIXED_FLOW"
   318	    return {"source": "v39a_descriptive_fallback", "value": {"bias": bias}}

   321	def _load_state(state_path: str | Path | None) -> tuple[dict[str, Any] | None, dict[str, Any]]:
   322	    if state_path is None:
   323	        return None, {"status": "DISABLED", "path": ""}
   324	    path = Path(state_path)
   325	    if not path.exists():
   326	        return None, {"status": "INITIALIZED", "path": str(path)}
   327	    try:
   328	        raw = json.loads(path.read_text(encoding="utf-8"))
   329	    except (OSError, json.JSONDecodeError) as exc:
   330	        return None, {"status": "INVALID", "path": str(path), "error": type(exc).__name__}
   331	    if not isinstance(raw, dict) or raw.get("state_schema_version") != STATE_SCHEMA_VERSION:
   332	        return None, {"status": "INCOMPATIBLE", "path": str(path)}
   333	    return raw, {"status": "LOADED", "path": str(path)}

   336	def _advance_state(*, previous_state, state_load, cutoff, structure, zones, market_state, quality):
   337	    previous_cutoff = _parse_ts((previous_state or {}).get("last_cutoff_ts"))
   338	    invalid_future = previous_cutoff is not None and previous_cutoff > cutoff
   339	    prior = None if invalid_future else previous_state
   340	    prior_date = _parse_ts((prior or {}).get("market_day_utc"))
   341	    day_changed = prior_date is not None and prior_date.date() != cutoff.date()
   342	    gap_minutes = None
   343	    if prior_date is not None:
   344	        gap_minutes = int((cutoff - previous_cutoff).total_seconds() // 60) if previous_cutoff else None
   345	    prior_zones = (prior or {}).get("liquidity_zones") or {}
   346	    current_zones = zones if zones.get("available") else prior_zones
   347	    carried = prior is not None
   348	    rollover_count = int((prior or {}).get("daily_boundary", {}).get("rollover_count", 0)) + (1 if day_changed else 0)
   349	    continuity = "INVALID_FUTURE_STATE" if invalid_future else "CONTINUOUS_CARRY_FORWARD" if carried else "INITIALIZED"
   350	    state_memory = {
   351	        "state_schema_version": STATE_SCHEMA_VERSION,
   352	        "load_status": state_load.get("status"),
   353	        "previous_cutoff_ts": _format_ts(previous_cutoff) if previous_cutoff is not None else "",
   354	        "carried_forward": bool(carried),
   355	        "continuity": continuity,
   356	        "gap_minutes": gap_minutes,
   357	        "daily_boundary": {
   358	            "utc_day": cutoff.date().isoformat(),
   359	            "changed": bool(day_changed),
   360	            "previous_utc_day": prior_date.date().isoformat() if prior_date is not None else "",
   361	            "rollover_count": rollover_count,
   362	            "state_carried_across_boundary": bool(day_changed and carried),
   363	        },
   364	        "state_source": "persisted_monitor_state" if carried else "initial_snapshot",
   365	        "invalid_future_state_discarded": bool(invalid_future),
   366	    }
   367	    next_state = {
   368	        "state_schema_version": STATE_SCHEMA_VERSION,
   369	        "last_cutoff_ts": _format_ts(cutoff),
   370	        "market_day_utc": cutoff.date().isoformat(),
   371	        "carry_forward_count": int((prior or {}).get("carry_forward_count", 0)) + (1 if carried else 0),
   372	        "daily_boundary": state_memory["daily_boundary"],
   373	        "structure": structure,
   374	        "liquidity_zones": current_zones,
   375	        "market_state": market_state,
   376	        "quality": quality,
   377	    }
   378	    return state_memory, next_state

   381	def _write_state(path: Path, state: Mapping[str, Any]) -> None:
   382	    path.parent.mkdir(parents=True, exist_ok=True)
   383	    fd, tmp_name = tempfile.mkstemp(prefix=f".{path.name}.", suffix=".tmp", dir=str(path.parent))
   384	    try:
   385	        with os.fdopen(fd, "w", encoding="utf-8") as handle:
   386	            json.dump(_json_safe(dict(state)), handle, indent=2, sort_keys=True)
   387	            handle.write("\n")
   388	            handle.flush()
   389	            os.fsync(handle.fileno())
   390	        os.replace(tmp_name, path)
   391	    finally:
   392	        if os.path.exists(tmp_name):
   393	            os.unlink(tmp_name)

   396	def _parse_ts(value: Any) -> pd.Timestamp | None:
   397	    if not value:
   398	        return None
   399	    ts = pd.Timestamp(value)
   400	    if ts.tzinfo is None:
   401	        return ts.tz_localize("UTC")
   402	    return ts.tz_convert("UTC")
   403	
   404	
   405	def _format_ts(value: Any) -> str:
   406	    ts = _parse_ts(value)
   407	    if ts is None:
   408	        raise MonitorSnapshotV39AError("timestamp is required")
   409	    return ts.isoformat().replace("+00:00", "Z")

   411	
   412	def _json_safe(value: Any) -> Any:
   413	    return json.loads(json.dumps(value, ensure_ascii=False, default=str))
