   156	def metrics(results, candidates):
   157	    filled=[r for r in results if r.entry_status=='FILLED']
   158	    closed=sorted([r for r in filled if r.lifecycle_class in RESOLVED],key=lambda r:r.exit_ts)
   159	    pnl=np.array([r.net_pnl_usdc for r in closed],dtype=float)
   160	    equity=np.r_[0,np.cumsum(pnl)]
   161	    losses=-float(pnl[pnl<0].sum())
   162	    monthly={}
   163	    for month in sorted({r.signal_ts_utc.strftime('%Y-%m') for r in results}):
   164	        rr=[r for r in closed if r.signal_ts_utc.strftime('%Y-%m')==month]
   165	        monthly[month]={'closed':len(rr),'net_usdc':sum(r.net_pnl_usdc for r in rr)}
   166	    return {'candidates':len(candidates),'filled':len(filled),'closed':len(closed),
   167	            'unresolved':len(filled)-len(closed),'net_usdc':float(pnl.sum()),
   168	            'gross_usdc':sum(r.gross_pnl_usdc for r in closed),
   169	            'commission_usdc':sum(r.commission_usdc for r in closed),
   170	            'slippage_usdc':sum(r.slippage_usdc for r in closed),
   171	            'profit_factor':float(pnl[pnl>0].sum())/losses if losses>0 else None,
   172	            'net_wins':int((pnl>0).sum()),'tp1_hits':sum(r.tp1_fill_ts is not None for r in closed),
   173	            'max_closed_equity_drawdown_usdc':float((np.maximum.accumulate(equity)-equity).max()),
   174	            'unresolved_realized_partial_net':sum(r.net_pnl_usdc or 0 for r in filled if r.lifecycle_class not in RESOLVED),
   175	            'status_counts':dict(Counter(r.entry_status for r in results)),
   176	            'lifecycle_counts':dict(Counter(r.lifecycle_class for r in results)),
   177	            'blocked_reasons':dict(Counter(r.blocked_reason for r in results if r.blocked_reason)),
   178	            'monthly_by_signal_month':monthly}

   180	def replay(out,label):
   181	    manifest=json.loads((out/'protocol_freeze.json').read_text())
   182	    if manifest['code_hashes']!=code_hashes():raise RuntimeError('Code changed since protocol freeze')
   183	    verify_inputs(out)
   184	    destination=out/label
   185	    destination.mkdir(exist_ok=False)
   186	    cfg=CONFIG if label=='base' else replace(CONFIG,exit_slippage_bps=2,stop_slippage_bps=4)
   187	    raw=json.loads((out/'candidates.json').read_text())
   188	    cs=[Candidate(**dict(r,signal_ts_utc=datetime.fromisoformat(r['signal_ts_utc']))) for r in raw]
   189	    cs=[c for c in cs if replay_eligible(c,manifest)]
   190	    signal=[b for b in load_legacy(BASE/'execution_feed/btcusdt_legacy_spot_1m/daily') if b.ts<=utc(manifest['outcome_end'])]
   191	    execution=load_feed(BASE/'execution_feed/btcusdc_spot_1m/daily',date_from='2026-03-19',date_to='2026-08-19',feed_role='official_spot_execution')
   192	    if any(b.ts-a.ts!=timedelta(minutes=1) for a,b in zip(execution,execution[1:])):
   193	        raise ValueError('Execution gap')
   194	    st=[b.ts for b in signal];et=[b.ts for b in execution]
   195	    results=[];events=[];errors=[]
   196	    for i,c in enumerate(cs):
   197	        try:
   198	            r,ev=replay_candidate(c,execution,cfg,reference_bars=signal,_timestamps=et,_reference_timestamps=st)
   199	        except BacktestContractError as exc:
   200	            r=TradeResult(trade_id='ERROR_'+c.candidate_id,candidate_id=c.candidate_id,experiment_id=cfg.experiment_id,
   201	                replay_mode='independent_opportunity',candidate_group=c.candidate_group,side=c.side,
   202	                signal_ts_utc=c.signal_ts_utc,entry_status='INVALID',lifecycle_class='INTEGRITY_ERROR',blocked_reason=str(exc))
   203	            ev=[]
   204	        if r.entry_status=='INVALID':errors.append({'candidate_id':c.candidate_id,'reason':r.blocked_reason})
   205	        results.append(r);events.extend(e.to_dict() for e in ev)
   206	        if (i+1)%20==0:print(label,'replayed',i+1,'/',len(cs),flush=True)
   207	    portfolio,pevents,_=replay_portfolio(cs,results,cfg)
   208	    rows(destination/'independent_trades.csv',(r.to_dict() for r in results))
   209	    rows(destination/'portfolio_trades.csv',(r.to_dict() for r in portfolio))
   210	    rows(destination/'execution_events.csv',events)
   211	    rows(destination/'portfolio_events.csv',(e.to_dict() for e in pevents))
   212	    report={'independent':metrics(results,cs),'portfolio':metrics(portfolio,cs),
   213	            'integrity_errors':errors,'config':cfg.to_dict()}
   214	    dump(destination/'metrics.json',report)
   215	    print(json.dumps(report,indent=2),flush=True)
