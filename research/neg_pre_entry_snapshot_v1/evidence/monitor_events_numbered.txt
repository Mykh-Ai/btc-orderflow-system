    99	def build_event_log(
   100	    *,
   101	    registry: pd.DataFrame,
   102	    feed: pd.DataFrame,
   103	    volume_delta_state: pd.DataFrame,
   104	    previous_registry: pd.DataFrame | None = None,
   105	) -> pd.DataFrame:
   106	    if registry.empty or feed.empty:
   107	        return pd.DataFrame(columns=EVENT_LOG_COLUMNS)
   108	
   109	    frame = feed.sort_values("Timestamp", kind="mergesort").reset_index(drop=True)
   110	    run_start = frame["Timestamp"].min()
   111	    run_end = frame["Timestamp"].max()
   112	    previous_status = _previous_status_map(previous_registry)
   113	    context = _context_map(volume_delta_state)
   114	    events: list[dict[str, object]] = []
   115	
   116	    for _, row in registry.iterrows():
   117	        zone = row.to_dict()
   118	        events.extend(_interaction_events(zone, frame, previous_status, context))
   119	        lifecycle_event = _registry_lifecycle_event(
   120	            zone, run_start, run_end, previous_status, context
   121	        )
   122	        if lifecycle_event is not None:
   123	            events.append(lifecycle_event)
   124	
   125	    if not events:
   126	        return pd.DataFrame(columns=EVENT_LOG_COLUMNS)
   127	    out = pd.DataFrame(events, columns=EVENT_LOG_COLUMNS)
   128	    out = out[out["event_type"].isin(ALLOWED_EVENT_TYPES)].copy()
   129	    out["_event_type_order"] = out["event_type"].map(EVENT_TYPE_ORDER).fillna(999)
   130	    out = out.sort_values(
   131	        ["event_timestamp", "zone_id", "_event_type_order", "event_type"],
   132	        kind="mergesort",
   133	    ).reset_index(drop=True)
   134	    out["event_id"] = [f"event_{idx + 1:06d}" for idx in range(len(out))]
   135	    out = assign_market_move_groups(out)
   136	    return out[EVENT_LOG_COLUMNS]

   139	def assign_market_move_groups(event_log: pd.DataFrame) -> pd.DataFrame:
   140	    out = event_log.copy()
   141	    if out.empty:
   142	        return pd.DataFrame(columns=EVENT_LOG_COLUMNS)
   143	    out["market_move_id"] = ""
   144	    out["market_move_role"] = "NONE"
   145	    out["market_move_event_count"] = 0
   146	    out["group_start_timestamp"] = ""
   147	    out["group_end_timestamp"] = ""
   148	    out["group_span_minutes"] = ""
   149	    out["grouping_window_mode"] = ""
   150	    unresolved = out[out["event_type"] == "LIQUIDITY_SWEEP_UNRESOLVED"].copy()
   151	    if unresolved.empty:
   152	        return out[EVENT_LOG_COLUMNS]
   153	
   154	    groups: list[list[int]] = []
   155	    for _, side_frame in unresolved.sort_values(
   156	        ["side", "event_timestamp", "zone_id", "event_id"], kind="mergesort"
   157	    ).groupby("side", sort=True):
   158	        current: list[int] = []
   159	        group_start_ts: pd.Timestamp | None = None
   160	        for idx, row in side_frame.iterrows():
   161	            event_ts = pd.Timestamp(row["event_timestamp"])
   162	            if (
   163	                current
   164	                and group_start_ts is not None
   165	                and event_ts - group_start_ts > pd.Timedelta(minutes=MARKET_MOVE_GROUP_WINDOW_MINUTES)
   166	            ):
   167	                groups.append(current)
   168	                current = []
   169	                group_start_ts = None
   170	            if not current:
   171	                group_start_ts = event_ts
   172	            current.append(idx)
   173	        if current:
   174	            groups.append(current)
   175	
   176	    groups.sort(
   177	        key=lambda idxs: (
   178	            pd.Timestamp(out.loc[idxs[0], "event_timestamp"]),
   179	            str(out.loc[idxs[0], "side"]),
   180	            str(out.loc[idxs[0], "event_id"]),
   181	        )
   182	    )
   183	    for group_number, idxs in enumerate(groups, start=1):
   184	        group = out.loc[idxs].copy()
   185	        first_ts = pd.Timestamp(group["event_timestamp"].map(pd.Timestamp).min())
   186	        last_ts = pd.Timestamp(group["event_timestamp"].map(pd.Timestamp).max())
   187	        span_minutes = _span_minutes(first_ts, last_ts)
   188	        side = str(group.iloc[0]["side"])
   189	        move_id = _market_move_id(first_ts, side, group_number)
   190	        primary_idx = _primary_event_index(group)
   191	        out.loc[idxs, "market_move_id"] = move_id
   192	        out.loc[idxs, "market_move_event_count"] = len(idxs)
   193	        out.loc[idxs, "market_move_role"] = "SECONDARY"
   194	        out.loc[idxs, "group_start_timestamp"] = _format_ts(first_ts)
   195	        out.loc[idxs, "group_end_timestamp"] = _format_ts(last_ts)
   196	        out.loc[idxs, "group_span_minutes"] = span_minutes
   197	        out.loc[idxs, "grouping_window_mode"] = GROUPING_WINDOW_MODE
   198	        out.loc[primary_idx, "market_move_role"] = "PRIMARY"
   199	    return out[EVENT_LOG_COLUMNS]

   202	def build_market_move_groups(event_log: pd.DataFrame) -> pd.DataFrame:
   203	    if event_log.empty or "market_move_id" not in event_log.columns:
   204	        return pd.DataFrame(columns=MARKET_MOVE_GROUP_COLUMNS)
   205	    grouped = event_log[
   206	        (event_log["event_type"] == "LIQUIDITY_SWEEP_UNRESOLVED")
   207	        & (event_log["market_move_id"].fillna("").astype(str) != "")
   208	    ].copy()
   209	    if grouped.empty:
   210	        return pd.DataFrame(columns=MARKET_MOVE_GROUP_COLUMNS)
   211	
   212	    rows = []
   213	    for _, group in grouped.groupby("market_move_id", sort=True):
   214	        rows.append(_market_move_group_row(group))
   215	    rows = sorted(
   216	        rows,
   217	        key=lambda row: (pd.Timestamp(row["event_timestamp"]), str(row["side"]), str(row["market_move_id"])),
   218	    )
   219	    return pd.DataFrame(rows, columns=MARKET_MOVE_GROUP_COLUMNS)

   232	            "unresolved_sweep": 0,

   234	            "unresolved_sweep_by_data_quality": "RAW=0, RECOVERED_DEGRADED=0",

   264	        **move_stats,
   265	    }

   268	def _interaction_events(
   269	    zone: dict[str, object],
   270	    feed: pd.DataFrame,
   271	    previous_status: dict[str, str],
   272	    context: dict[str, dict[str, float]],
   273	) -> list[dict[str, object]]:
   274	    if str(zone.get("status", "")) in {"MERGED", "EXPIRED"}:
   275	        return []
   276	    first_seen = pd.Timestamp(zone["first_seen_at"])
   277	    run_slice = feed[feed["Timestamp"] >= first_seen].copy()
   278	    if run_slice.empty:
   279	        return []
   280	    if _consumed_before_or_at(zone, run_slice["Timestamp"].min()):
   281	        return []
   282	
   283	    cross_row = _first_cross(zone, run_slice)
   284	    touch_row = _first_touch(zone, run_slice)
   285	    if cross_row is not None:
   286	        cross_event = _market_event(
   287	            zone=zone,
   288	            event_type="LIQUIDITY_ZONE_CROSSED_UNCLASSIFIED",
   289	            candle=cross_row,
   290	            feed=feed,
   291	            previous_status=previous_status,
   292	            context=context,
   293	            trigger="price_crossed_zone_far_side",
   294	            excursion_abs=_cross_excursion(zone, cross_row),
   295	            new_status="CROSSED_UNCLASSIFIED",
   296	        )
   297	        events = [cross_event]
   298	        unresolved_event = _unresolved_sweep_event(
   299	            zone=zone,
   300	            candle=cross_row,
   301	            feed=feed,
   302	            previous_status=previous_status,
   303	            context=context,
   304	        )
   305	        if unresolved_event is not None:
   306	            events.append(unresolved_event)
   307	        return events
   308	
   309	    if touch_row is not None:
   310	        return [
   311	            _market_event(
   312	                zone=zone,
   313	                event_type="LIQUIDITY_ZONE_TOUCHED",
   314	                candle=touch_row,
   315	                feed=feed,
   316	                previous_status=previous_status,
   317	                context=context,
   318	                trigger="price_entered_zone_range",
   319	                excursion_abs=0,
   320	                new_status="TOUCHED",
   321	            )
   322	        ]
   323	
   324	    approach_row = _first_approach(zone, run_slice)
   325	    if approach_row is not None:
   326	        return [
   327	            _market_event(
   328	                zone=zone,
   329	                event_type="LIQUIDITY_ZONE_APPROACHED",
   330	                candle=approach_row,
   331	                feed=feed,
   332	                previous_status=previous_status,
   333	                context=context,
   334	                trigger="price_reached_approach_threshold_without_touch",
   335	                excursion_abs=0,
   336	                new_status="APPROACHED",
   337	            )
   338	        ]
   339	    return []

   342	def _unresolved_sweep_event(
   343	    *,
   344	    zone: dict[str, object],
   345	    candle: pd.Series,
   346	    feed: pd.DataFrame,
   347	    previous_status: dict[str, str],
   348	    context: dict[str, dict[str, float]],
   349	) -> dict[str, object] | None:
   350	    event_timestamp = pd.Timestamp(candle["Timestamp"])
   351	    first_seen_at = pd.Timestamp(zone["first_seen_at"])
   352	    if first_seen_at >= event_timestamp:
   353	        return None
   354	    if _consumed_before_or_at(zone, event_timestamp):
   355	        return None
   356	    if str(zone.get("precision_status", "")) == PRECISION_TOO_WIDE:
   357	        return None
   358	    if _non_fresh_role_before_or_at(zone, event_timestamp):
   359	        return None
   360	
   361	    if _is_repeated_prior_cross_without_new_transition(
   362	        zone=zone,
   363	        candle=candle,
   364	        feed=feed,
   365	        previous_status=previous_status,
   366	    ):
   367	        return None
   368	
   369	    excursion_abs = _cross_excursion(zone, candle)
   370	    min_excursion_abs = _min_sweep_excursion(zone)
   371	    if excursion_abs <= 0 or excursion_abs < min_excursion_abs:
   372	        return None
   373	
   374	    event_ts = _format_ts(event_timestamp)
   375	    ctx = context.get(
   376	        event_ts,
   377	        {"volume_zscore": 0.0, "delta_zscore": 0.0, "oi_change": 0.0},
   378	    )
   379	    data_quality = _event_data_quality(zone, candle)
   380	    activity_evidence = _activity_evidence(ctx, data_quality)
   381	    if not activity_evidence:
   382	        return None
   383	
   384	    evidence = {
   385	        "activity_evidence": activity_evidence,
   386	        "activity_passed": True,
   387	        "candidate_evidence_status": "SUFFICIENT_UNRESOLVED",
   388	        "confidence_score": int(float(zone.get("confidence_score", 0) or 0)),
   389	        "confidence_tier": str(zone.get("confidence_tier", "")),
   390	        "data_quality": data_quality,
   391	        "delta_zscore": float(ctx["delta_zscore"]),
   392	        "event_class": "LIQUIDITY_SWEEP_UNRESOLVED",
   393	        "event_close": float(candle["ClosePrice"]),
   394	        "event_high": float(candle["HiPrice"]),
   395	        "event_low": float(candle["LowPrice"]),
   396	        "event_timestamp": event_ts,
   397	        "excursion_abs": float(excursion_abs),
   398	        "excursion_passed": True,
   399	        "first_seen_at": _format_ts(first_seen_at),
   400	        "min_excursion_abs": float(min_excursion_abs),
   401	        "oi_change": float(ctx["oi_change"]),
   402	        "pre_existing_zone": True,
   403	        "price_lower": float(zone["price_lower"]),
   404	        "price_mid": float(zone["price_mid"]),
   405	        "price_upper": float(zone["price_upper"]),
   406	        "reaction_status": "UNRESOLVED",
   407	        "side": str(zone["side"]),
   408	        "active_forward": str(zone.get("active_forward", "")),
   409	        "active_forward_role": str(zone.get("active_forward_role", "")),
   410	        "accepted_above_at": str(zone.get("accepted_above_at", "")),
   411	        "accepted_below_at": str(zone.get("accepted_below_at", "")),
   412	        "consumption_status": str(zone.get("consumption_status", "")),
   413	        "consumed_at": str(zone.get("consumed_at", "")),
   414	        "failed_acceptance_count": int(float(zone.get("failed_acceptance_count", 0) or 0)),
   415	        "first_sweep_at": str(zone.get("first_sweep_at", "")),
   416	        "resweep_count": int(float(zone.get("resweep_count", 0) or 0)),
   417	        "source_timeframes": str(zone.get("source_timeframes", "")),
   418	        "structural_zone_mode": str(zone.get("structural_zone_mode", "")),
   419	        "volume_zscore": float(ctx["volume_zscore"]),
   420	        "zone_behavior_state": str(zone.get("zone_behavior_state", "")),
   421	        "zone_id": str(zone["zone_id"]),
   422	        "zone_outer_lower": float(zone.get("zone_outer_lower", zone["price_lower"]) or zone["price_lower"]),
   423	        "zone_outer_upper": float(zone.get("zone_outer_upper", zone["price_upper"]) or zone["price_upper"]),
   424	        "zone_core_lower": float(zone.get("zone_core_lower", zone["price_lower"]) or zone["price_lower"]),
   425	        "zone_core_upper": float(zone.get("zone_core_upper", zone["price_upper"]) or zone["price_upper"]),
   426	        "zone_type": str(zone["zone_type"]),
   427	        **_score_instrumentation_evidence(zone),
   428	    }
   429	    return {
   430	        "event_id": "",
   431	        "event_timestamp": event_ts,
   432	        "event_type": "LIQUIDITY_SWEEP_UNRESOLVED",
   433	        "zone_id": zone["zone_id"],
   434	        "side": zone["side"],
   435	        "price_before": _price_before(feed, event_timestamp),
   436	        "event_high": float(candle["HiPrice"]),
   437	        "event_low": float(candle["LowPrice"]),
   438	        "event_close": float(candle["ClosePrice"]),
   439	        "excursion_abs": float(excursion_abs),
   440	        "excursion_atr": 0,
   441	        "volume_zscore": ctx["volume_zscore"],
   442	        "delta_zscore": ctx["delta_zscore"],
   443	        "oi_change": ctx["oi_change"],
   444	        "reaction_status": "UNRESOLVED",
   445	        "evidence_json": _json(evidence),
   446	        "data_quality": data_quality,
   447	    }

   450	def _registry_lifecycle_event(
   451	    zone: dict[str, object],
   452	    run_start: pd.Timestamp,
   453	    run_end: pd.Timestamp,
   454	    previous_status: dict[str, str],
   455	    context: dict[str, dict[str, float]],
   456	) -> dict[str, object] | None:
   457	    status = str(zone.get("status", ""))
   458	    if status not in {"MERGED", "EXPIRED"}:
   459	        return None
   460	    prior_status = previous_status.get(str(zone["zone_id"]), "")
   461	    if prior_status == status:
   462	        return None
   463	    event_ts = pd.Timestamp(zone.get("last_updated_at") or zone.get("last_seen_at"))
   464	    if event_ts < run_start or event_ts > run_end + pd.Timedelta(days=1):
   465	        return None
   466	    event_type = (
   467	        "LIQUIDITY_ZONE_MERGED"
   468	        if status == "MERGED"
   469	        else "LIQUIDITY_ZONE_EXPIRED"
   470	    )
   471	    evidence = _base_evidence(
   472	        zone=zone,
   473	        previous_status=prior_status,
   474	        new_status=status,
   475	        trigger="registry_lifecycle_update",
   476	        candle_timestamp=_format_ts(event_ts),
   477	        candle_high=0,
   478	        candle_low=0,
   479	        candle_close=0,
   480	        context={"volume_zscore": 0.0, "delta_zscore": 0.0, "oi_change": 0.0},
   481	    )
   482	    if status == "MERGED":
   483	        evidence["merged_from_zone_id"] = str(zone["zone_id"])
   484	        evidence["merged_into_zone_id"] = str(zone.get("merged_into_zone_id", ""))
   485	    else:
   486	        evidence["age_days"] = int(float(zone.get("age_days", 0) or 0))
   487	        evidence["max_zone_age_days"] = 7
   488	        evidence["invalidation_reason"] = str(zone.get("invalidation_reason", ""))
   489	    return {
   490	        "event_id": "",
   491	        "event_timestamp": _format_ts(event_ts),
   492	        "event_type": event_type,
   493	        "zone_id": zone["zone_id"],
   494	        "side": zone["side"],

   614	def _first_cross(zone: dict[str, object], feed: pd.DataFrame) -> pd.Series | None:
   615	    lower = float(zone.get("zone_outer_lower", zone["price_lower"]) or zone["price_lower"])
   616	    upper = float(zone.get("zone_outer_upper", zone["price_upper"]) or zone["price_upper"])
   617	    if zone["side"] == "BUY_SIDE":
   618	        crossed = feed[feed["HiPrice"] > upper]
   619	    else:
   620	        crossed = feed[feed["LowPrice"] < lower]
   621	    if crossed.empty:
   622	        return None
   623	    return crossed.iloc[0]

   638	def _first_approach(zone: dict[str, object], feed: pd.DataFrame) -> pd.Series | None:
   639	    latest_close = float(feed.iloc[-1]["ClosePrice"])
   640	    threshold = max(APPROACH_THRESHOLD_MIN_USD, latest_close * APPROACH_THRESHOLD_FRACTION)
   641	    lower = float(zone.get("zone_outer_lower", zone["price_lower"]) or zone["price_lower"])
   642	    upper = float(zone.get("zone_outer_upper", zone["price_upper"]) or zone["price_upper"])
   643	    if zone["side"] == "BUY_SIDE":
   644	        approached = feed[(feed["HiPrice"] >= lower - threshold) & (feed["HiPrice"] < lower)]
   645	    else:
   646	        approached = feed[(feed["LowPrice"] <= upper + threshold) & (feed["LowPrice"] > upper)]
   647	    if approached.empty:
   648	        return None
   649	    return approached.iloc[0]

   902	def _primary_selection_diagnostics(group: pd.DataFrame, primary_idx: int) -> tuple[str, dict[str, object]]:
   903	    scored = group.copy()
   904	    scored["_confidence_score"] = scored.apply(_event_confidence_score, axis=1)
   905	    scored["_zone_width_pct"] = scored.apply(_event_zone_width_pct, axis=1)
   906	    selected = scored.loc[primary_idx]
   907	    reason = "highest_confidence_score"
   908	    tie_breakers: list[str] = []
   909	
   910	    confidence_peers = scored[scored["_confidence_score"] == selected["_confidence_score"]]
   911	    if len(confidence_peers) > 1:
   912	        tie_breakers.append("highest_excursion_abs")
   913	        excursion_peers = confidence_peers[
   914	            pd.to_numeric(confidence_peers["excursion_abs"], errors="coerce")
   915	            == float(selected["excursion_abs"])
   916	        ]
   917	        reason = "highest_excursion_abs_tiebreak"
   918	        if len(excursion_peers) > 1:
   919	            tie_breakers.append("narrowest_zone")
   920	            width_peers = excursion_peers[
   921	                excursion_peers["_zone_width_pct"] == selected["_zone_width_pct"]
   922	            ]
   923	            reason = "narrowest_zone_tiebreak"
   924	            if len(width_peers) > 1:
   925	                tie_breakers.append("zone_id")
   926	                zone_peers = width_peers[
   927	                    width_peers["zone_id"].astype(str) == str(selected["zone_id"])
   928	                ]
   929	                reason = "zone_id_tiebreak"
   930	                if len(zone_peers) > 1:

   951	    evidence = _event_evidence(row)
   952	    value = _float_or_none(evidence.get("zone_width_pct", ""))
   953	    if value is not None:
   954	        return value
   955	    lower = _float_or_none(evidence.get("price_lower", ""))
   956	    upper = _float_or_none(evidence.get("price_upper", ""))
   957	    mid = _float_or_none(evidence.get("price_mid", ""))
   958	    if lower is None or upper is None or not mid:
   959	        return float("inf")
   960	    return abs(upper - lower) / abs(mid) * 100
